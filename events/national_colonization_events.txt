namespace = national_colonization

# Castile and Spain enter the New World through the Caribbean basin. The AI
# strategy separately delays its strong preference for mainland Mexico.
national_colonization.1 = {
	type = country_event

	title = national_colonization.1.t
	desc = national_colonization.1.d
	flavor = national_colonization.1.f

	event_image = { texture = "gfx/event_pictures/discovery.dds" }
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = never_fire_again_modifier_time }
	placement = ROOT

	trigger = {
		has_technology_researched = colonization
		is_subject = no
		OR = {
			c:CAS ?= this
			c:SPA ?= this
			was_formed_from = CAS
		}
	}

	option = {
		name = national_colonization.1.a
		default_option = yes
		ai_chance = { base = 100 }
		set_variable = aoc_spanish_caribbean_phase
		add_involvement = {
			strategic_region = sr:region_central_america
			value = 2000
		}
		add_modifier = { name = colonial_rush years = 10 }
		activate_law = law_type:law_colonial_resettlement
	}

	option = {
		name = national_colonization.1.b
		ai_chance = { base = 0 }
	}
}

# Spain: Caribbean foothold -> mainland -> northern South America -> southern cone.
national_colonization.4 = {
	type = country_event
	title = national_colonization.4.t
	desc = national_colonization.4.d
	flavor = national_colonization.4.f
	event_image = { texture = "gfx/event_pictures/discovery.dds" }
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = never_fire_again_modifier_time }
	placement = ROOT

	trigger = {
		has_technology_researched = colonization
		OR = {
			has_variable = aoc_spanish_caribbean_phase
			c:CAS ?= this
			c:SPA ?= this
			was_formed_from = CAS
		}
		NOT = { has_variable = aoc_spanish_mainland_phase }
		any_state = {
			count >= 2
			is_in_geographic_region = geographic_region_aoc_spanish_caribbean
			owner = { OR = { this = root is_subject_of = root } }
		}
	}

	option = {
		name = national_colonization.4.a
		default_option = yes
		set_variable = aoc_spanish_caribbean_phase
		set_variable = aoc_spanish_mainland_phase
	}
}
