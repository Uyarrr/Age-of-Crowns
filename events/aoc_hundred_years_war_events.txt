aoc_hyw.320 = {
	type = country_event
	title = aoc_hyw_summons_title
	desc = aoc_hyw_summons_desc
	icon = "gfx/interface/icons/event_icons/hundred_years_war.dds"
	event_image = { texture = "gfx/event_pictures/hundred_years_war.dds" }
	duration = 1
	trigger = {
		exists = scope:aoc_hyw_round_play
		exists = var:aoc_hyw_round_crown
	}
	immediate = {
		var:aoc_hyw_round_crown = { save_scope_as = aoc_hyw_summoning_crown }
	}
	option = {
		name = aoc_hyw_summons_accept
		trigger = {
			exists = scope:aoc_hyw_round_play
			aoc_hyw_can_answer_summons = yes
		}
		ai_chance = {
			base = 65
			modifier = { trigger = { is_subject_of = scope:aoc_hyw_summoning_crown } add = 30 }
			modifier = { trigger = { is_at_war = yes } add = -45 }
			modifier = { trigger = { in_default = yes } add = -40 }
		}
		set_variable = aoc_hyw_answered_summons
		scope:aoc_hyw_round_play ?= {
			if = {
				limit = {
					is_war = yes
					war ?= { NOT = { is_war_participant = root } }
				}
				war = { join_war = { target = root side = scope:aoc_hyw_summoning_crown } }
			}
			else_if = {
				limit = { is_war = no scope:aoc_hyw_summoning_crown = c:ENG }
				add_initiator_backers = { root }
			}
			else_if = {
				limit = { is_war = no }
				add_target_backers = { root }
			}
		}
	}
	option = {
		name = aoc_hyw_summons_refuse
		default_option = yes
		ai_chance = { base = 35 }
		if = {
			limit = { exists = scope:aoc_hyw_round_play aoc_hyw_can_answer_summons = yes }
			save_scope_as = aoc_hyw_refusing_country
			change_relations = { country = scope:aoc_hyw_summoning_crown value = -75 }
			if = {
				limit = { aoc_hyw_crown_or_subject = no }
				aoc_hyw_withdraw_support = yes
			}
			scope:aoc_hyw_summoning_crown = { trigger_event = { id = aoc_hyw.321 popup = yes } }
		}
	}
}

			NAME = aoc_hyw_autonomy_treaty
			ARTICLE = money_transfer
			MONTHS = 12
			RATE = 1000
		}
	}
	option = {
		name = aoc_hyw_autonomy_reject
		default_option = yes
		ai_chance = { base = 55 }
		scope:aoc_hyw_petitioner ?= {
			add_liberty_desire = 10

			aoc_hyw_pending_invite_matches = yes

			}

		}
		ai_chance = {
			base = 40
			modifier = { trigger = { "relations(scope:aoc_hyw_inviting_crown)" >= 75 } add = 40 }
			modifier = { trigger = { is_at_war = yes } add = -30 }
		}
		set_variable = { name = aoc_hyw_pledged_crown value = scope:aoc_hyw_inviting_crown }
		remove_variable = aoc_hyw_pending_invite
	}
	option = {
		name = aoc_hyw_invitation_decline
		default_option = yes
		ai_chance = { base = 60 }
		if = {
			limit = { aoc_hyw_pending_invite_matches = yes }

		}
	}
}
