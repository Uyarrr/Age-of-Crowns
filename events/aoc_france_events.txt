aoc_france.5 = {
	type = country_event
	title = aoc_france_cloth_of_gold.title
	desc = aoc_france_cloth_of_gold.desc
	flavor = aoc_france_cloth_of_gold.flavor

	event_image = { texture = "gfx/event_pictures/joust.dds" }
	icon = "gfx/interface/icons/invention_icons/france_coa.dds"
	fire_only_once = yes
	duration = 3

	trigger = {
		c:FRA ?= this
		aoc_era_research_mid = { ERA = 2 }
		aoc_era_event_window_open = { ERA = 2 }
		is_subject = no
		any_country = {
			aoc_english_successor = yes
			aoc_france_can_negotiate_with = { OTHER = root }
		}
	}

	immediate = {
		set_variable = aoc_content_seen_aoc_france_5

		save_scope_as = aoc_france_envoys
		if = {
			limit = { c:ENG ?= { aoc_france_can_negotiate_with = { OTHER = root } } }
			c:ENG = { save_scope_as = aoc_france_cloth_of_gold_guest }
		}
		else = {
			ordered_country = {
				limit = {
					aoc_english_successor = yes
					aoc_france_can_negotiate_with = { OTHER = root }
				}
				order_by = prestige
				max = 1
				save_scope_as = aoc_france_cloth_of_gold_guest
			}
		}
	}

	option = {
		name = aoc_france_cloth_of_gold.a
		default_option = yes
		ai_chance = { base = 80 }
		trigger = {
			scope:aoc_france_cloth_of_gold_guest ?= {
				aoc_english_successor = yes
				aoc_france_can_negotiate_with = { OTHER = root }
			}
		}
		scope:aoc_france_cloth_of_gold_guest = { trigger_event = { id = aoc_france.10 popup = yes } }
	}

	option = {
		name = aoc_france_cloth_of_gold.b
		ai_chance = { base = 20 }
		add_modifier = { name = aoc_france_court_splendor years = 10 }
	}
}
