GLOBAL = { 	
    every_country = {
 		add_modifier = {
			name = organic_development
			years = 5
		}
    }

    # Starting Supply Ships from Vanilla
    every_country = {
		limit = {
			navy_size >= 1
		}
		add_supply_ships = {
			value = 1
			add = navy_size
			if = {
				limit = {
					any_scope_building = {
						is_building_type = building_shipyard
					}
				}
				every_scope_building = {
					limit = {
						is_building_type = building_shipyard
					}
					add = 10
				}
			}
			multiply = 8
		}
	}
}
