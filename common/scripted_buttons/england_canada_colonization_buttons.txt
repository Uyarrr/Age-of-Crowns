# England's Canada-only alternative to the generic colonial administration.
# These are the six British colonial subjects with Canadian territory in 1836.
# Columbia District uses ORG, matching vanilla's dynamic British subject name.

je_england_canada_new_brunswick = {
	name = "je_england_canada_new_brunswick"
	desc = "je_england_canada_new_brunswick_desc"
	visible = { NOT = { exists = c:NBS } }
	possible = {
		NOT = { exists = c:NBS }
		has_technology_researched = colonization
		owns_entire_state_region = STATE_NEW_BRUNSWICK
		is_at_war = no
		is_diplomatic_play_committed_participant = no
	}
	ai_chance = { value = 100 }
	effect = {
		save_scope_as = england_canada_overlord
		religion = { save_scope_as = england_canada_overlord_religion }
		create_country = {
			tag = NBS
			origin = root
			province = p:xC91FD7
			province = p:xEE2FFF
			province = p:x5A0CA8
			province = p:x71A015
			province = p:x326F36
			province = p:xF01C37
			province = p:x70A0CE
			province = p:x484A09
			province = p:x082153
			province = p:x62FC6A
			province = p:xDE0FD5
			province = p:xBB0F05
			province = p:x7040C0
			province = p:x4D9EEE
			province = p:x8ED253
			province = p:xD090A6
			province = p:x20CF66
			province = p:x66DFE7
			province = p:x493CD4
			province = p:x740012
			province = p:x8E553E
			province = p:x70C040
			on_created = {
				save_scope_as = england_canada_new_subject
				england_canada_inherit_overlord_identity_effect = yes
				set_variable = england_canada_new_brunswick_established
				activate_law = law_type:law_colonial_administration
				add_modifier = { name = colonial_administration }
				every_scope_state = { effect_destroy_colonial_office = yes }
			}
		}
		hidden_effect = {
			create_diplomatic_pact = { country = c:NBS type = colony }
		}
	}
}

je_england_canada_hudsons_bay = {
	name = "je_england_canada_hudsons_bay"
	desc = "je_england_canada_hudsons_bay_desc"
	visible = {
		OR = {
			NOT = { exists = c:HBC }
			c:HBC ?= { is_subject_of = root }
		}
	}
	possible = {
		has_technology_researched = colonization
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		OR = {
			# Found the company from any controlled interior state.
			AND = {
				NOT = { exists = c:HBC }
				any_scope_state = {
					OR = {
						state_region = s:STATE_MANITOBA
						state_region = s:STATE_SASKATCHEWAN
						state_region = s:STATE_ALBERTA
						state_region = s:STATE_NORTHWEST_TERRITORIES
						state_region = s:STATE_YUKON_TERRITORY
						state_region = s:STATE_NUNAVUT
					}
				}
			}
			# Once founded, every later use transfers all newly controlled interior states.
			AND = {
				exists = c:HBC
				c:HBC ?= { is_subject_of = root }
				any_scope_state = {
					OR = {
						state_region = s:STATE_MANITOBA
						state_region = s:STATE_SASKATCHEWAN
						state_region = s:STATE_ALBERTA
						state_region = s:STATE_NORTHWEST_TERRITORIES
						state_region = s:STATE_YUKON_TERRITORY
						state_region = s:STATE_NUNAVUT
					}
				}
			}
		}
	}
	ai_chance = { value = 100 }
	effect = {
		save_scope_as = england_canada_overlord
		religion = { save_scope_as = england_canada_overlord_religion }
		if = {
			limit = { exists = c:HBC }
			every_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_MANITOBA
						state_region = s:STATE_SASKATCHEWAN
						state_region = s:STATE_ALBERTA
						state_region = s:STATE_NORTHWEST_TERRITORIES
						state_region = s:STATE_YUKON_TERRITORY
						state_region = s:STATE_NUNAVUT
					}
				}
				effect_destroy_colonial_office = yes
				set_state_owner = c:HBC
			}
		}
		else = {
			random_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_MANITOBA
						state_region = s:STATE_SASKATCHEWAN
						state_region = s:STATE_ALBERTA
						state_region = s:STATE_NORTHWEST_TERRITORIES
						state_region = s:STATE_YUKON_TERRITORY
						state_region = s:STATE_NUNAVUT
					}
				}
				save_scope_as = england_canada_hudsons_bay_founding_state
			}
			create_country = {
				tag = HBC
				origin = root
				state = scope:england_canada_hudsons_bay_founding_state
				on_created = {
					save_scope_as = england_canada_new_subject
					england_canada_inherit_overlord_identity_effect = yes
					set_variable = england_canada_hudsons_bay_established
					activate_law = law_type:law_colonial_administration
					activate_law = law_type:law_frontier_colonization
					add_modifier = { name = colonial_administration }
					every_scope_state = { effect_destroy_colonial_office = yes }
				}
			}
			hidden_effect = {
				create_diplomatic_pact = { country = c:HBC type = chartered_company }
			}
			every_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_MANITOBA
						state_region = s:STATE_SASKATCHEWAN
						state_region = s:STATE_ALBERTA
						state_region = s:STATE_NORTHWEST_TERRITORIES
						state_region = s:STATE_YUKON_TERRITORY
						state_region = s:STATE_NUNAVUT
					}
				}
				effect_destroy_colonial_office = yes
				set_state_owner = c:HBC
			}
		}
	}
}
