aoc_hyw_claim_throne = {
	name = aoc_hyw_claim_throne
	desc = aoc_hyw_claim_throne_desc
	visible = { c:ENG ?= this }
	possible = {
		aoc_hyw_peacetime_action = yes
		NOT = { has_variable = aoc_hyw_claimed_throne }
		NOT = { has_global_variable = aoc_hyw_campaign_started }
		scope:journal_entry = {
			var:aoc_hyw_english_wins = 0
			var:aoc_hyw_french_wins = 0
			var:aoc_hyw_stalemates = 0
		}
	}
	effect = {
		set_variable = aoc_hyw_claimed_throne
		change_infamy = 75
		add_modifier = { name = aoc_hyw_throne_claim_infamy }
	}
	ai_chance = {
		base = 0
		modifier = {
			trigger = { army_power_projection > c:FRA.army_power_projection }
			add = 50
		}
	}
}
