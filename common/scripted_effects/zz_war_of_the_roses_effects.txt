# The Crown IG follows the throne, while a displaced monarch returns to the noble estate.
wotr_prepare_crown_accession_effect = {
	ruler ?= {
		set_interest_group = PREV.ig:ig_landowners
	}
}

wotr_ensure_henry_vi_effect = {
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = wotr_henry_vi_character
				}
			}
		}
		create_character = {
			template = ENG_henry_vi_lancaster_wotr
			on_created = {
				set_variable = wotr_henry_vi_character
			}
		}
	}
}

wotr_install_lancaster_ruler_effect = {
	wotr_prepare_crown_accession_effect = yes
	if = {
		limit = { NOT = { has_variable = wotr_readeption_event } }
		wotr_ensure_henry_vi_effect = yes
		random_scope_character = {
			limit = { has_variable = wotr_henry_vi_character }
			set_interest_group = PREV.ig:ig_crown
			set_character_as_ruler = yes
		}
	}
	else = {
		wotr_ensure_edward_westminster_effect = yes
		random_scope_character = {
			limit = { has_variable = wotr_edward_westminster_character }
			set_interest_group = PREV.ig:ig_crown
			set_character_as_ruler = yes
		}
	}
	wotr_set_lancaster_dynasty_effect = yes
}

wotr_record_war_fought_effect = {
	# Both the incumbent and a victorious claimant receive the same breathing room.
	# Country variables mirror the bars so the next monthly assessment agrees.
	remove_modifier = war_of_the_roses_high_tension
	set_variable = { name = war_of_roses_tension_current_var value = 25 }
	je:je_england_2 ?= {
		set_bar_progress = { name = je_war_of_the_roses_tension_bar value = 25 }
	}
	if = {
		limit = { has_variable = wotr_dynasty_lancaster }
		set_variable = { name = war_of_roses_balance_current_var value = 70 }
		je:je_england_2 ?= { set_bar_progress = { name = je_war_of_the_roses_balance_bar value = 70 } }
	}
	else_if = {
		limit = { has_variable = wotr_dynasty_york }
		set_variable = { name = war_of_roses_balance_current_var value = 30 }
		je:je_england_2 ?= { set_bar_progress = { name = je_war_of_the_roses_balance_bar value = 30 } }
	}
	if = {
		limit = { has_variable = wotr_wars_fought_var }
		change_variable = { name = wotr_wars_fought_var add = 1 }
	}
	else = {
		set_variable = { name = wotr_wars_fought_var value = 1 }
	}
	set_variable = { name = wotr_phase_var value = 4 }
	set_variable = { name = wotr_stability_months_var value = 0 }
	set_variable = { name = war_of_roses_tension_offset_var value = -15 }
}
