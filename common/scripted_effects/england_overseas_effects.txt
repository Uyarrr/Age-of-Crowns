# Called in the new subject's scope after create_country. Inherit the sponsor's
# cultures/religion without replacing indigenous cultures or generating free pops.
aoc_england_overseas_identity_effect = {
 save_scope_as = aoc_england_new_subject
 set_state_religion = scope:aoc_england_overlord_religion
 scope:aoc_england_overlord = {
  every_primary_culture = {
   limit = { NOT = { is_primary_culture_of = scope:aoc_england_new_subject } }
   save_temporary_scope_as = aoc_england_overlord_culture
   scope:aoc_england_new_subject = { add_primary_culture = scope:aoc_england_overlord_culture }
  }
 }
 activate_law = law_type:law_colonial_administration
 activate_law = law_type:law_frontier_colonization
 add_modifier = { name = colonial_administration }
 every_scope_state = { effect_destroy_colonial_office = yes }
}
# Record a meaningful growth target once, including when adopting an existing Cape.
aoc_england_cape_baseline_effect = {
 c:SAF ?= {
  if = {
   limit = { is_subject_of = root NOT = { has_variable = aoc_england_cape_gdp_target } }
   set_variable = { name = aoc_england_cape_gdp_target value = gdp }
   change_variable = { name = aoc_england_cape_gdp_target multiply = 1.25 }
  }
 }
}
# Country scope. Aggregate only the Crown's American possessions and its subjects.
aoc_england_new_world_census_effect = {
 set_variable = { name = aoc_england_new_world_population value = 0 }
 set_variable = { name = aoc_england_new_world_gdp value = 0 }
 every_state = {
  limit = { aoc_england_new_world_state = yes }
  root = {
   change_variable = { name = aoc_england_new_world_population add = prev.state_population }
   change_variable = { name = aoc_england_new_world_gdp add = prev.gdp }
  }
 }
}

# Reconcile charter slots when a subject is lost or regained.
aoc_england_charter_company_slots_effect = {
 if = {
  limit = {
   has_variable = aoc_england_east_india_chartered
   OR = {
    AND = {
     aoc_england_overseas_country = yes
     c:BIC ?= { is_subject_of = root is_subject_type = subject_type_chartered_company }
    }
    AND = {
     aoc_french_east_india_sponsor = yes
     c:FIC ?= { is_subject_of = root is_subject_type = subject_type_chartered_company }
    }
   }
  }
  if = { limit = { NOT = { has_modifier = aoc_england_east_india_company_slot } } add_modifier = { name = aoc_england_east_india_company_slot } }
 }
 else = { remove_modifier = aoc_england_east_india_company_slot }
 if = {
  limit = { has_variable = aoc_england_hudson_bay_chartered exists = c:HBC c:HBC ?= { is_subject_of = root is_subject_type = subject_type_chartered_company } }
  if = { limit = { NOT = { has_modifier = aoc_england_hudson_bay_company_slot } } add_modifier = { name = aoc_england_hudson_bay_company_slot } }
 }
 else = { remove_modifier = aoc_england_hudson_bay_company_slot }
}

# Country scope. The charter variables retain their original keys for existing campaigns.
aoc_charter_east_india = {
  save_scope_as = aoc_england_overlord
  religion = { save_scope_as = aoc_england_overlord_religion }
  if = {
   limit = { NOT = { exists = c:$TAG$ } }
   random_scope_state = { limit = { is_in_geographic_region = geographic_region_india } save_scope_as = aoc_england_charter_state }
   create_country = {
    tag = $TAG$
    origin = root
    state = scope:aoc_england_charter_state
    on_created = {
     aoc_england_overseas_identity_effect = yes
     set_country_type = company
     activate_law = law_type:law_presidential_republic
     activate_law = law_type:law_oligarchy
    }
   }
   create_diplomatic_pact = { country = c:$TAG$ type = chartered_company }
  }
  every_scope_state = {
   limit = { is_in_geographic_region = geographic_region_india }
   effect_destroy_colonial_office = yes
   set_state_owner = c:$TAG$
  }

 c:$TAG$ = { aoc_copy_country_identity_effect = { SOURCE = scope:aoc_england_overlord } }
 set_variable = aoc_england_east_india_chartered
 aoc_england_charter_company_slots_effect = yes
 if = { limit = { NOT = { has_company = company_type:$COMPANY$ } } add_company = company_type:$COMPANY$ }
 company:$COMPANY$ = { add_owned_country = c:$TAG$ }

  c:$TAG$ = {
   if = {
    limit = { root = { NOT = { has_variable = aoc_england_east_india_endowed } } }
    root = { set_variable = aoc_england_east_india_endowed }
    set_variable = aoc_england_east_india_endowed
    capital = {
     create_building = { building = building_textile_mill level = 2 reserves = 0.5 }
     create_building = { building = building_cotton_plantation add_ownership = { company = { type = $COMPANY$ country = scope:aoc_england_overlord levels = 2 } } reserves = 0.5 }
    }
    random_scope_state = {
     limit = { is_coastal = yes }
     create_building = { building = building_port level = 1 reserves = 0.5 activate_production_methods = { pm_building_port_main_1 } }
    }
   }
  }

  s:STATE_GUJARAT = { add_claim = c:$TAG$ }
  s:STATE_BOMBAY = { add_claim = c:$TAG$ }
  s:STATE_MADRAS = { add_claim = c:$TAG$ }
  s:STATE_WEST_BENGAL = { add_claim = c:$TAG$ }
  s:STATE_EAST_BENGAL = { add_claim = c:$TAG$ }
  s:STATE_CIRCARS = { add_claim = c:$TAG$ }
  custom_tooltip = aoc_england_charter_east_india_tt
}
