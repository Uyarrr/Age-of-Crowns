aoc_is_castilian_or_spanish_subject = {
	OR = {
		AND = {
			exists = c:SPA
			is_direct_subject_of = c:SPA
		}
		AND = {
			exists = c:CAS
			is_direct_subject_of = c:CAS
		}
	}
}
