	OR = {
		AND = {
			exists = c:SPA
			is_direct_subject_of = c:SPA
		}
		AND = {
			exists = c:CAS
			is_direct_subject_of = c:CAS
		}
	}
}

# Australia and New Zealand use fixed tags as well as generated countries.
# Require a colonial identity so ordinary local subjects keep their own names.
aoc_is_australian_colonial_nation = {
	aoc_is_or_was_colonial_nation = yes
	capital = { is_in_geographic_region = geographic_region_aoc_england_australia }
}
