# Country scope unless noted. ROOT is always the journal/button owner.
aoc_england_overseas_country = {
 OR = { c:ENG ?= this was_formed_from = ENG }
 is_subject = no
 is_revolutionary = no
 is_secessionist = no
}
# Country scope: direct Crown ownership or any tier of its subject hierarchy.
aoc_england_overseas_owner = { OR = { this = root is_subject_of = root } }
# State scope.
aoc_england_new_world_state = {
 is_in_geographic_region = geographic_region_americas
 owner = { aoc_england_overseas_owner = yes }
}
aoc_england_cape_station_ready = {
 exists = c:SAF
 c:SAF ?= {
  is_subject_of = root
  any_scope_state = { state_region = s:STATE_CAPE_COLONY }
  total_population >= 25000
  gdp >= 100000
  any_scope_state = { is_coastal = yes has_building = building_port }
 }
}
aoc_england_cape_developed = {
 exists = c:SAF
 c:SAF ?= {
  is_subject_of = root
  owns_entire_state_region = STATE_CAPE_COLONY
  total_population >= 150000
  gdp >= 750000
  gdp >= var:aoc_england_cape_gdp_target
  average_sol >= 10
 }
}
# Country scope. Each charter has its own country tag and company type.
aoc_east_india_sponsor = {
	OR = {
		aoc_england_overseas_country = yes
		aoc_french_east_india_sponsor = yes
	}
}

aoc_french_east_india_sponsor = {
	c:FRA ?= this
	is_subject = no
	is_revolutionary = no
	is_secessionist = no
}

aoc_east_india_unlocked = {
	OR = {
		AND = {
			aoc_england_overseas_country = yes
			custom_tooltip = {
				text = aoc_england_india_unlocked_tt
				has_variable = aoc_england_india_unlocked
			}
		}
		AND = {
			aoc_french_east_india_sponsor = yes
			has_technology_researched = france_east_india_company
		}
	}
}

aoc_east_india_charter_possible = {
	OR = {
		NOT = { exists = c:$TAG$ }
		c:$TAG$ ?= {
			is_subject_of = root
			is_subject_type = subject_type_chartered_company
		}
	}
	OR = {
		any_scope_state = { is_in_geographic_region = geographic_region_india }
		c:$TAG$ ?= {
			is_subject_of = root
			NOT = { has_variable = aoc_england_east_india_endowed }
		}
	}
}

aoc_east_india_presidencies = {
	has_company = company_type:$COMPANY$
	c:$TAG$ ?= {
		is_subject_of = root
		is_subject_type = subject_type_chartered_company
		total_population >= 250000
		gdp >= 500000
	}
	# Count distinct state regions rather than fragments of the same region.
	any_state_region = {
		count >= 3
		is_in_geographic_region = geographic_region_india
		any_scope_state = { owner = c:$TAG$ num_provinces >= 1 }
	}
}

aoc_england_naval_leadership = {
 naval_power_projection > 0
 NOT = { any_country = { count >= 3 naval_power_projection > root.naval_power_projection } }
 "goods_production_rank(g:navalsupplies)" <= 3
 any_scope_state = {
  any_scope_building = { is_building_type = building_shipyard occupancy >= 0.5 }
 }
}
