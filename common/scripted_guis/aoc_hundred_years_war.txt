aoc_hyw_influence_gui = {
	scope = country
	saved_scopes = { aoc_hyw_target }
	is_shown = { c:ENG ?= this }
	is_valid = {
		aoc_hyw_can_influence = yes
		scope:aoc_hyw_target ?= { aoc_hyw_influence_target = yes }
	}
	effect = {
		if = {
			limit = {
				aoc_hyw_can_influence = yes
				scope:aoc_hyw_target ?= { aoc_hyw_influence_target = yes }
			}
			aoc_hyw_influence_subject = yes
		}
	}
}

		scope:aoc_hyw_target ?= { aoc_hyw_appanage_target = yes }
	}
	effect = {

	}
