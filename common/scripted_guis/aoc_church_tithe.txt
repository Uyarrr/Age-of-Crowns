aoc_church_issue_relief_gui = {
	scope = country
	saved_scopes = { aoc_relief_target }

	is_shown = {
		c:PAP ?= this
	}

	is_valid = {
		aoc_church_can_issue_relief = yes
		scope:aoc_relief_target ?= { aoc_church_relief_eligible = yes }
	}

	effect = {
		if = {
			limit = {
				aoc_church_can_issue_relief = yes
			scope:aoc_relief_target ?= { aoc_church_relief_eligible = yes }
			}
			scope:aoc_relief_target = { aoc_church_issue_relief = yes }
		}
	}
}
