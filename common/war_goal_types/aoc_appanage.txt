REPLACE:increase_autonomy = {
	icon = "gfx/interface/icons/war_goals/increase_autonomy.dds"

	kind = increase_autonomy

	settings = {
		require_target_be_part_of_war

		# Validation
		validate_conflicts_war_goals_all
		validate_subject_relation
		validate_increase_autonomy
	}

	execution_priority = 40

	contestion_type = control_all_own_states

	target_type = country

	possible = {
		NOT = { is_subject_type = subject_type_appanage }
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		NOT = { is_subject_type = subject_type_appanage }
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		root.overlord = scope:target_country
	}

	fill_per_week = 2
	deplete_per_week = 2

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy = {
		value = 0
	}

	on_enforced = {
		# script effect on top of the predefined code effect
	}

	mirrored_wargoal = {
		method = swap
		type = reduce_autonomy
	}

	ai = {
		is_significant_demand = no
	}
}

REPLACE:reduce_autonomy = {
	icon = "gfx/interface/icons/war_goals/reduce_autonomy.dds"

	kind = reduce_autonomy

	settings = {
		require_target_be_part_of_war

		# Validation
		conflicts_with_make_subject
		validate_subject_relation
		validate_conflicts_war_goals_all
		validate_conflicts_make_subject
	}

	execution_priority = 39

	contestion_type = control_any_target_incorporated_state

	target_type = country

	possible = {
		scope:target_country = { NOT = { is_subject_type = subject_type_appanage } }
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		scope:target_country = { NOT = { is_subject_type = subject_type_appanage } }
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		scope:target_country.overlord = root
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy_severity = 0.2

	infamy = {
		value = 0

		scope:target_country = {
			add = country_infamy_value
		}

		multiply = {
			desc = "INFAMY_FROM_LOYAL_SUBJECT"
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REDUCE_AUTONOMY_LOW_LIBERTY_DESIRE_MULT
			}
		}

		multiply = {
			desc = "INFAMY_FROM_REBELLIOUS_SUBJECT" # Rebellious as in they are the aggressor in the play
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						is_diplomatic_play_enemy_of = root
						scope:diplomatic_play ?= {
							initiator_is = scope:target_country
						}
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REBELLING_SUBJECT_MULT
			}
		}
	}

	on_enforced = {
		# script effect on top of the predefined code effect
	}

	mirrored_wargoal = {
		method = swap
		type = increase_autonomy
	}

	ai = {
		is_significant_demand = no
	}
}
