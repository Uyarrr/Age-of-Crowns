REPLACE:conquer_state = {
	icon = "gfx/interface/icons/war_goals/conquer_state.dds"

	kind = conquer_state

	settings = {
		require_target_be_part_of_war
		annexes_entire_state
		can_add_for_other_country
		requires_interest

		# Validation
		validate_target_not_treaty_port
		validate_conflicts_conquer_state

		conflicts_with_country_creation
		conflicts_with_annex_state
	}

	execution_priority = 69

	contestion_type = control_target_state

	target_type = state

	possible = {
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		scope:target_state = {
			owner = scope:target_country
			NOT = { has_claim_by = root }
		}
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 15
		}
		add = {
			desc = "MANEUVERS_TARGET_STATE_POPULATION_FACTOR"
			# The surcharge reaches its cap near the 90th percentile of 1444 state populations.
			value = scope:target_state.state_population
			multiply = 15
			divide = 550000
			max = 15
		}
	}

	infamy = {
		value = 0

		scope:target_state = {
			save_temporary_scope_as = infamy_ignores_claim
			add = state_infamy_value
		}

		multiply = {
			desc = "INFAMY_FROM_LOYAL_SUBJECT"
			value = 1
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_SUBJECT_MULT
			}
		}

		multiply = {
			desc = "INFAMY_FROM_REBELLIOUS_SUBJECT" # Rebellious as in they are the aggressor in the play
			value = 1
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						is_diplomatic_play_enemy_of = root
						scope:diplomatic_play ?= {
							initiator_is = scope:target_country
						}
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REBELLING_SUBJECT_MULT
			}
		}
	}

	on_enforced = {
		# script effect on top of the predefined code effect
	}

	mirrored_wargoal = {
		method = territorial
	}

	ai = {
		is_significant_demand = yes
	}
}
