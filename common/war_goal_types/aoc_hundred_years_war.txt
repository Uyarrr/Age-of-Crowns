	kind = custom
	settings = { require_target_be_part_of_war }
	execution_priority = 18
	contestion_type = control_half_target_country_states
	target_type = country
	possible = { always = no }
	valid = { scope:diplomatic_play ?= { is_diplomatic_play_type = dp_aoc_hyw } }

			limit = { c:FRA ?= this }
			set_global_variable = aoc_hyw_french_round_victory
		}
	}

	ai = { is_significant_demand = yes }
