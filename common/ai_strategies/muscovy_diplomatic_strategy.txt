ai_strategy_muscovy_form_russia = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = no
	}

	undesirable_infamy_level = { value = 60 }
	unacceptable_infamy_level = { value = 100 }

	# Muscovy cares relatively little about ideological differences while
	# consolidating the Russian lands.
	ideological_opinion_effect_mult = { value = 0.25 }

	wargoal_maneuvers_fraction = { value = 0.35 }

	# Keep this at 0 rather than relying on negative neutrality values.
	# High boldness and target scoring already make Muscovy expansionist.
	diplomatic_play_neutrality = { value = 0 }

	diplomatic_play_boldness = { value = 85 }

	recklessness = { value = 1.0 }


	# ---------------------------------------------------------------------
	# AGGRESSION
	# ---------------------------------------------------------------------
	#
	# PRIMARY OBJECTIVE:
	# Consolidate the Russian heartlands required to form Russia.
	#
	# Military SIZE is handled by the global military AI script.
	#

	aggression = {
		value = 0.20


		# -------------------------------------------------------------
		# CORE RUSSIAN CONSOLIDATION TARGETS
		# -------------------------------------------------------------
		#
		# These states are the main formation / western Rus priorities.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country = {
					OR = {
						has_state_in_state_region = STATE_NOVGOROD
						has_state_in_state_region = STATE_TVER
						has_state_in_state_region = STATE_YAROSLAVL
						has_state_in_state_region = STATE_PSKOV
						has_state_in_state_region = STATE_SMOLENSK
						has_state_in_state_region = STATE_RYAZAN
						has_state_in_state_region = STATE_ORYOL
						has_state_in_state_region = STATE_KURSK
						has_state_in_state_region = STATE_TAMBOV
						has_state_in_state_region = STATE_VYATKA
						has_state_in_state_region = STATE_ARKHANGELSK
						has_state_in_state_region = STATE_KHARKOV
					}
				}
			}

			add = 4.0


			# Major historical Rus consolidation targets.
			if = {
				limit = {
					scope:target_country = {
						OR = {
							c:NOV ?= this
							c:TVR ?= this
							c:RYZ ?= this
							c:U67 ?= this
							c:ODO ?= this
						}
					}
				}

				add = 1.5
			}


			# Adjacent targets are easier and more strategically sensible.
			if = {
				limit = {
					has_strategic_adjacency = scope:target_country
				}

				add = 0.75
			}


			# Prefer weaker targets.
			if = {
				limit = {
					scope:target_country.country_rank < root.country_rank
				}

				add = 0.50
			}


			# Exploit enemies already occupied by another war.
			if = {
				limit = {
					scope:target_country = {
						is_in_war = yes
					}
				}

				add = 0.50
			}


			# Isolated countries are easier consolidation targets.
			if = {
				limit = {
					scope:target_country = {
						num_allies < 1
					}
				}

				add = 0.25
			}
		}


		# -------------------------------------------------------------
		# OTHER RUSSIAN HOMELANDS
		# -------------------------------------------------------------
		#
		# Still desirable, but noticeably below the immediate western
		# consolidation targets.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country = {
					any_scope_state = {
						is_homeland_of_country_cultures = root
					}

					NOR = {
						has_state_in_state_region = STATE_NOVGOROD
						has_state_in_state_region = STATE_TVER
						has_state_in_state_region = STATE_YAROSLAVL
						has_state_in_state_region = STATE_PSKOV
						has_state_in_state_region = STATE_SMOLENSK
						has_state_in_state_region = STATE_RYAZAN
						has_state_in_state_region = STATE_ORYOL
						has_state_in_state_region = STATE_KURSK
						has_state_in_state_region = STATE_TAMBOV
						has_state_in_state_region = STATE_VYATKA
						has_state_in_state_region = STATE_ARKHANGELSK
						has_state_in_state_region = STATE_KHARKOV
					}
				}
			}

			add = 2.0
		}


		# -------------------------------------------------------------
		# NEIGHBORING SMALL POWERS
		# -------------------------------------------------------------
		#
		# After / alongside consolidation, neighboring minor countries
		# remain reasonable targets for frontier security or subjugation.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				has_strategic_adjacency = scope:target_country

				scope:target_country.country_rank <= rank_value:minor_power
			}

			add = 0.75
		}


		# -------------------------------------------------------------
		# SUBJECTS
		# -------------------------------------------------------------
		#
		# Subjects should normally be integrated through autonomy /
		# annexation mechanics rather than attacked directly.
		#

		if = {
			limit = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			multiply = 0.02
		}


		# -------------------------------------------------------------
		# DISTANT / IRRELEVANT COUNTRIES
		# -------------------------------------------------------------
		#
		# Muscovy should not randomly launch overseas or distant wars
		# while still pursuing Russian consolidation.
		#

		if = {
			limit = {
				NOT = {
					has_strategic_adjacency = scope:target_country
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							is_homeland_of_country_cultures = root
						}
					}
				}
			}

			multiply = 0.05
		}


		max = 7.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL WEIGHTS
	# ---------------------------------------------------------------------

	secret_goal_weights = {
		conquer = 5.0
		dominate = 2.0

		protect = 0.50

		befriend = 0.10
		reconcile = 0.0
	}


	# ---------------------------------------------------------------------
	# WARGOAL WEIGHTS
	# ---------------------------------------------------------------------

	wargoal_weights = {
		conquer_state = 5.0
		return_state = 4.0

		make_protectorate = 2.0
		make_dominion = 0.50

		reduce_autonomy = 4.0
		annex_country = 5.0

		transfer_subject = 1.0

		force_nationalization = 0.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL SCORING
	# ---------------------------------------------------------------------

	secret_goal_scores = {


		# -------------------------------------------------------------
		# CONQUER RUSSIAN HOMELANDS
		# -------------------------------------------------------------

		conquer = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country = {
					any_scope_state = {
						is_homeland_of_country_cultures = root
					}
				}
			}

			score = {
				value = 500


				# Immediate Russian consolidation states receive
				# a very large additional priority.
				if = {
					limit = {
						scope:target_country = {
							OR = {
								has_state_in_state_region = STATE_NOVGOROD
								has_state_in_state_region = STATE_TVER
								has_state_in_state_region = STATE_YAROSLAVL
								has_state_in_state_region = STATE_PSKOV
								has_state_in_state_region = STATE_SMOLENSK
								has_state_in_state_region = STATE_RYAZAN
								has_state_in_state_region = STATE_ORYOL
								has_state_in_state_region = STATE_KURSK
								has_state_in_state_region = STATE_TAMBOV
								has_state_in_state_region = STATE_VYATKA
								has_state_in_state_region = STATE_ARKHANGELSK
								has_state_in_state_region = STATE_KHARKOV
							}
						}
					}

					add = 300
				}


				# Major Rus rivals receive another substantial bonus.
				if = {
					limit = {
						scope:target_country = {
							OR = {
								c:NOV ?= this
								c:TVR ?= this
								c:RYZ ?= this
								c:U67 ?= this
								c:ODO ?= this
							}
						}
					}

					add = 250
				}


				if = {
					limit = {
						has_strategic_adjacency = scope:target_country
					}

					add = 150
				}


				if = {
					limit = {
						scope:target_country = {
							is_in_war = yes
						}
					}

					add = 100
				}


				if = {
					limit = {
						scope:target_country = {
							num_allies < 1
						}
					}

					add = 100
				}
			}
		}


		# -------------------------------------------------------------
		# DOMINATE NEIGHBORING SMALL POWERS
		# -------------------------------------------------------------

		dominate = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				has_strategic_adjacency = scope:target_country

				scope:target_country.country_rank <= rank_value:minor_power

				# Russian homelands should normally be taken directly,
				# not converted into protectorates.
				NOT = {
					scope:target_country = {
						any_scope_state = {
							is_homeland_of_country_cultures = root
						}
					}
				}
			}

			score = {
				value = 175


				if = {
					limit = {
						scope:target_country = {
							num_allies < 1
						}
					}

					add = 75
				}


				if = {
					limit = {
						scope:target_country = {
							is_in_war = yes
						}
					}

					add = 75
				}
			}
		}


		# -------------------------------------------------------------
		# PROTECT SUBJECTS
		# -------------------------------------------------------------

		protect = {
			trigger = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			score = {
				value = 100
			}
		}


		# -------------------------------------------------------------
		# IGNORE IRRELEVANT DISTANT COUNTRIES
		# -------------------------------------------------------------

		none = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				NOT = {
					has_strategic_adjacency = scope:target_country
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							is_homeland_of_country_cultures = root
						}
					}
				}
			}

			score = {
				value = 1000
			}
		}
	}


	# ---------------------------------------------------------------------
	# WARGOAL SCORING
	# ---------------------------------------------------------------------

	wargoal_scores = {


		# -------------------------------------------------------------
		# CONQUER STATE
		# -------------------------------------------------------------

		conquer_state = {

			# Russian cultural homelands are naturally valuable.
			if = {
				limit = {
					scope:target_state = {
						is_homeland_of_country_cultures = root
					}
				}

				add = 500
			}


			# Core western / Russian consolidation states receive
			# a much stronger priority.
			if = {
				limit = {
					scope:target_state = {
						OR = {
							state_region = s:STATE_NOVGOROD
							state_region = s:STATE_TVER
							state_region = s:STATE_YAROSLAVL
							state_region = s:STATE_PSKOV
							state_region = s:STATE_SMOLENSK
							state_region = s:STATE_RYAZAN
							state_region = s:STATE_ORYOL
							state_region = s:STATE_KURSK
							state_region = s:STATE_TAMBOV
							state_region = s:STATE_VYATKA
							state_region = s:STATE_ARKHANGELSK
							state_region = s:STATE_KHARKOV
						}
					}
				}

				add = 600
			}


			# Prefer contiguous expansion.
			if = {
				limit = {
					scope:target_state = {
						any_neighbouring_state = {
							owner = root
						}
					}
				}

				add = 150
			}


			# ---------------------------------------------------------
			# SIBERIAN DEPRIORITIZATION
			# ---------------------------------------------------------
			#
			# These should not distract Muscovy from consolidating
			# western Russia early.
			#

			if = {
				limit = {
					scope:target_state = {
						OR = {
							state_region = s:STATE_OB
							state_region = s:STATE_SURGUT
							state_region = s:STATE_TOMSK
							state_region = s:STATE_YAKUTSK
						}
					}
				}

				add = -500
			}
		}


		# -------------------------------------------------------------
		# RETURN STATE
		# -------------------------------------------------------------

		return_state = {
			if = {
				limit = {
					scope:target_state = {
						is_homeland_of_country_cultures = root
					}
				}

				add = 700
			}


			if = {
				limit = {
					scope:target_state = {
						OR = {
							state_region = s:STATE_NOVGOROD
							state_region = s:STATE_TVER
							state_region = s:STATE_YAROSLAVL
							state_region = s:STATE_PSKOV
							state_region = s:STATE_SMOLENSK
							state_region = s:STATE_RYAZAN
							state_region = s:STATE_ORYOL
							state_region = s:STATE_KURSK
							state_region = s:STATE_TAMBOV
							state_region = s:STATE_VYATKA
							state_region = s:STATE_ARKHANGELSK
							state_region = s:STATE_KHARKOV
						}
					}
				}

				add = 300
			}
		}


		# -------------------------------------------------------------
		# PROTECTORATE
		# -------------------------------------------------------------

		make_protectorate = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}
				}

				add = 225
			}
		}


		# -------------------------------------------------------------
		# SUBJECT INTEGRATION
		# -------------------------------------------------------------

		reduce_autonomy = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				add = 400
			}
		}


		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				add = 600
			}
		}
	}


	# ---------------------------------------------------------------------
	# STATE VALUE
	# ---------------------------------------------------------------------

	state_value = {
		value = 0


		# Russian cultural homeland.
		if = {
			limit = {
				scope:target_state = {
					is_homeland_of_country_cultures = root
				}
			}

			add = 150
		}


		# Main Russian consolidation territory.
		if = {
			limit = {
				scope:target_state = {
					OR = {
						state_region = s:STATE_NOVGOROD
						state_region = s:STATE_TVER
						state_region = s:STATE_YAROSLAVL
						state_region = s:STATE_PSKOV
						state_region = s:STATE_SMOLENSK
						state_region = s:STATE_RYAZAN
						state_region = s:STATE_ORYOL
						state_region = s:STATE_KURSK
						state_region = s:STATE_TAMBOV
						state_region = s:STATE_VYATKA
						state_region = s:STATE_ARKHANGELSK
						state_region = s:STATE_KHARKOV
					}
				}
			}

			add = 250
		}


		# Contiguous states are more strategically useful.
		if = {
			limit = {
				scope:target_state = {
					any_neighbouring_state = {
						owner = root
					}
				}
			}

			add = 75
		}


		# Siberian territory is deliberately lower priority while
		# Muscovy is still consolidating Russia.
		if = {
			limit = {
				scope:target_state = {
					OR = {
						state_region = s:STATE_OB
						state_region = s:STATE_SURGUT
						state_region = s:STATE_TOMSK
						state_region = s:STATE_YAKUTSK
					}
				}
			}

			add = -125
		}


		min = 0
		max = 500
	}


	# ---------------------------------------------------------------------
	# AVAILABILITY
	# ---------------------------------------------------------------------

	possible = {
		is_subject = no

		c:MUS ?= this
	}


	# ---------------------------------------------------------------------
	# STRATEGY SELECTION
	# ---------------------------------------------------------------------

	weight = {
		value = 1000
	}
}
