ai_strategy_england_unite_britain = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = yes
	}

	undesirable_infamy_level = { value = 65 }
	unacceptable_infamy_level = { value = 95 }

	ideological_opinion_effect_mult = { value = 0.60 }

	wargoal_maneuvers_fraction = { value = 0.45 }

	diplomatic_play_neutrality = { value = 0 }
	diplomatic_play_boldness = { value = 70 }

	recklessness = { value = 1.15 }


	# ---------------------------------------------------------------------
	# AGGRESSION
	# ---------------------------------------------------------------------
	#
	# England strongly prioritizes completing control over the British Isles.
	# Military SIZE itself is handled by the global military AI script.
	#

	aggression = {
		value = 0.25

		# Strong aggression toward countries holding territory
		# considered part of the British Isles.
		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country = {
					OR = {
						has_state_in_state_region = STATE_HOME_COUNTIES
						has_state_in_state_region = STATE_MIDLANDS
						has_state_in_state_region = STATE_EAST_ANGLIA
						has_state_in_state_region = STATE_WEST_COUNTRY
						has_state_in_state_region = STATE_YORKSHIRE
						has_state_in_state_region = STATE_LANCASHIRE
						has_state_in_state_region = STATE_WALES
						has_state_in_state_region = STATE_LOWLANDS
						has_state_in_state_region = STATE_HIGHLANDS
						has_state_in_state_region = STATE_ULSTER
						has_state_in_state_region = STATE_LEINSTER
						has_state_in_state_region = STATE_CONNAUGHT
						has_state_in_state_region = STATE_MUNSTER
					}
				}
			}

			add = 5.0
		}


		# While Britain is incomplete, England should be less distracted
		# by unrelated foreign expansion.
		if = {
			limit = {
				OR = {
					NOT = { owns_entire_state_region = STATE_HOME_COUNTIES }
					NOT = { owns_entire_state_region = STATE_MIDLANDS }
					NOT = { owns_entire_state_region = STATE_EAST_ANGLIA }
					NOT = { owns_entire_state_region = STATE_WEST_COUNTRY }
					NOT = { owns_entire_state_region = STATE_YORKSHIRE }
					NOT = { owns_entire_state_region = STATE_LANCASHIRE }
					NOT = { owns_entire_state_region = STATE_WALES }
					NOT = { owns_entire_state_region = STATE_LOWLANDS }
					NOT = { owns_entire_state_region = STATE_HIGHLANDS }
					NOT = { owns_entire_state_region = STATE_ULSTER }
					NOT = { owns_entire_state_region = STATE_LEINSTER }
					NOT = { owns_entire_state_region = STATE_CONNAUGHT }
					NOT = { owns_entire_state_region = STATE_MUNSTER }
				}

				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				scope:target_country = {
					NOR = {
						has_state_in_state_region = STATE_HOME_COUNTIES
						has_state_in_state_region = STATE_MIDLANDS
						has_state_in_state_region = STATE_EAST_ANGLIA
						has_state_in_state_region = STATE_WEST_COUNTRY
						has_state_in_state_region = STATE_YORKSHIRE
						has_state_in_state_region = STATE_LANCASHIRE
						has_state_in_state_region = STATE_WALES
						has_state_in_state_region = STATE_LOWLANDS
						has_state_in_state_region = STATE_HIGHLANDS
						has_state_in_state_region = STATE_ULSTER
						has_state_in_state_region = STATE_LEINSTER
						has_state_in_state_region = STATE_CONNAUGHT
						has_state_in_state_region = STATE_MUNSTER
					}
				}
			}

			add = -1.5
		}


		# Subjects are handled mainly through autonomy reduction / annexation,
		# rather than ordinary aggression.
		if = {
			limit = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			multiply = 0.10
		}
	}


	# ---------------------------------------------------------------------
	# SECRET GOALS
	# ---------------------------------------------------------------------

	secret_goal_weights = {
		conquer = 4.0
		dominate = 2.0
		protect = 0.5
		befriend = 0.0
		reconcile = 0.0
	}


	# ---------------------------------------------------------------------
	# WARGOAL PREFERENCES
	# ---------------------------------------------------------------------

	wargoal_weights = {
		conquer_state = 5.0
		return_state = 2.5

		make_protectorate = 2.5
		make_dominion = 0.5

		reduce_autonomy = 4.0
		annex_country = 5.0

		force_nationalization = 0.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL SCORING
	# ---------------------------------------------------------------------

	secret_goal_scores = {

		# Direct conquest is strongly preferred against countries
		# controlling territory in the British Isles.
		conquer = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country = {
					OR = {
						has_state_in_state_region = STATE_HOME_COUNTIES
						has_state_in_state_region = STATE_MIDLANDS
						has_state_in_state_region = STATE_EAST_ANGLIA
						has_state_in_state_region = STATE_WEST_COUNTRY
						has_state_in_state_region = STATE_YORKSHIRE
						has_state_in_state_region = STATE_LANCASHIRE
						has_state_in_state_region = STATE_WALES
						has_state_in_state_region = STATE_LOWLANDS
						has_state_in_state_region = STATE_HIGHLANDS
						has_state_in_state_region = STATE_ULSTER
						has_state_in_state_region = STATE_LEINSTER
						has_state_in_state_region = STATE_CONNAUGHT
						has_state_in_state_region = STATE_MUNSTER
					}
				}
			}

			score = {
				value = 750

				# Scotland is an especially important unification target.
				if = {
					limit = {
						scope:target_country = {
							c:SCO ?= this
						}
					}

					add = 250
				}

				# Irish territories receive an additional priority.
				if = {
					limit = {
						scope:target_country = {
							OR = {
								has_state_in_state_region = STATE_ULSTER
								has_state_in_state_region = STATE_LEINSTER
								has_state_in_state_region = STATE_CONNAUGHT
								has_state_in_state_region = STATE_MUNSTER
							}
						}
					}

					add = 200
				}
			}
		}


		# Smaller independent states in Scotland or Ireland can also
		# be brought under control through subject relationships.
		dominate = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				can_reach_target_country = yes

				scope:target_country.country_rank <= rank_value:minor_power

				scope:target_country = {
					OR = {
						has_state_in_state_region = STATE_LOWLANDS
						has_state_in_state_region = STATE_HIGHLANDS
						has_state_in_state_region = STATE_ULSTER
						has_state_in_state_region = STATE_LEINSTER
						has_state_in_state_region = STATE_CONNAUGHT
						has_state_in_state_region = STATE_MUNSTER
					}
				}
			}

			score = {
				value = 250
			}
		}


		# Existing British-Isles subjects are strategically important
		# and should be protected until they can be integrated.
		protect = {
			trigger = {
				scope:target_country = {
					is_subject_of = root
				}

				scope:target_country = {
					OR = {
						has_state_in_state_region = STATE_LOWLANDS
						has_state_in_state_region = STATE_HIGHLANDS
						has_state_in_state_region = STATE_ULSTER
						has_state_in_state_region = STATE_LEINSTER
						has_state_in_state_region = STATE_CONNAUGHT
						has_state_in_state_region = STATE_MUNSTER
					}
				}
			}

			score = {
				value = 150
			}
		}
	}


	# ---------------------------------------------------------------------
	# WARGOAL SCORING
	# ---------------------------------------------------------------------

	wargoal_scores = {

		conquer_state = {

			# Increase conquest preference against countries already
			# selected for conquest.
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = conquer
					}
				}

				add = 300
			}


			# British Isles states are extremely valuable conquest targets.
			if = {
				limit = {
					scope:target_state = {
						OR = {
							state_region = s:STATE_HOME_COUNTIES
							state_region = s:STATE_MIDLANDS
							state_region = s:STATE_EAST_ANGLIA
							state_region = s:STATE_WEST_COUNTRY
							state_region = s:STATE_YORKSHIRE
							state_region = s:STATE_LANCASHIRE
							state_region = s:STATE_WALES
							state_region = s:STATE_LOWLANDS
							state_region = s:STATE_HIGHLANDS
							state_region = s:STATE_ULSTER
							state_region = s:STATE_LEINSTER
							state_region = s:STATE_CONNAUGHT
							state_region = s:STATE_MUNSTER
						}
					}
				}

				add = 600
			}
		}


		return_state = {
			if = {
				limit = {
					scope:target_state = {
						OR = {
							state_region = s:STATE_HOME_COUNTIES
							state_region = s:STATE_MIDLANDS
							state_region = s:STATE_EAST_ANGLIA
							state_region = s:STATE_WEST_COUNTRY
							state_region = s:STATE_YORKSHIRE
							state_region = s:STATE_LANCASHIRE
							state_region = s:STATE_WALES
							state_region = s:STATE_LOWLANDS
							state_region = s:STATE_HIGHLANDS
							state_region = s:STATE_ULSTER
							state_region = s:STATE_LEINSTER
							state_region = s:STATE_CONNAUGHT
							state_region = s:STATE_MUNSTER
						}
					}
				}

				add = 400
			}
		}


		make_protectorate = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}
				}

				add = 200
			}
		}


		reduce_autonomy = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}

					scope:target_country = {
						OR = {
							has_state_in_state_region = STATE_LOWLANDS
							has_state_in_state_region = STATE_HIGHLANDS
							has_state_in_state_region = STATE_ULSTER
							has_state_in_state_region = STATE_LEINSTER
							has_state_in_state_region = STATE_CONNAUGHT
							has_state_in_state_region = STATE_MUNSTER
						}
					}
				}

				add = 450
			}
		}


		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}

					scope:target_country = {
						OR = {
							has_state_in_state_region = STATE_LOWLANDS
							has_state_in_state_region = STATE_HIGHLANDS
							has_state_in_state_region = STATE_ULSTER
							has_state_in_state_region = STATE_LEINSTER
							has_state_in_state_region = STATE_CONNAUGHT
							has_state_in_state_region = STATE_MUNSTER
						}
					}
				}

				add = 700
			}
		}
	}


	# ---------------------------------------------------------------------
	# STATE VALUE
	# ---------------------------------------------------------------------
	#
	# British Isles states receive an additional strategic value.
	#

	state_value = {
		value = 0

		if = {
			limit = {
				scope:target_state = {
					OR = {
						state_region = s:STATE_HOME_COUNTIES
						state_region = s:STATE_MIDLANDS
						state_region = s:STATE_EAST_ANGLIA
						state_region = s:STATE_WEST_COUNTRY
						state_region = s:STATE_YORKSHIRE
						state_region = s:STATE_LANCASHIRE
						state_region = s:STATE_WALES
						state_region = s:STATE_LOWLANDS
						state_region = s:STATE_HIGHLANDS
						state_region = s:STATE_ULSTER
						state_region = s:STATE_LEINSTER
						state_region = s:STATE_CONNAUGHT
						state_region = s:STATE_MUNSTER
					}
				}
			}

			add = 175
		}

		min = 0
		max = 200
	}


	# ---------------------------------------------------------------------
	# AVAILABILITY
	# ---------------------------------------------------------------------

	possible = {
		is_subject = no

		OR = {
			c:ENG ?= this
			c:GBR ?= this
		}
	}


	# ---------------------------------------------------------------------
	# STRATEGY SELECTION WEIGHT
	# ---------------------------------------------------------------------

	weight = {
		value = 500

		# Become much more likely to choose this strategy when
		# important parts of Britain or Ireland remain outside our control.
		if = {
			limit = {
				OR = {
					NOT = { owns_entire_state_region = STATE_WALES }
					NOT = { owns_entire_state_region = STATE_LOWLANDS }
					NOT = { owns_entire_state_region = STATE_HIGHLANDS }
					NOT = { owns_entire_state_region = STATE_ULSTER }
					NOT = { owns_entire_state_region = STATE_LEINSTER }
					NOT = { owns_entire_state_region = STATE_CONNAUGHT }
					NOT = { owns_entire_state_region = STATE_MUNSTER }
				}
			}

			add = 500
		}

		# Britain may still use the strategy to recover lost parts
		# of the British Isles after formation.
		if = {
			limit = {
				c:GBR ?= this
			}

			add = 100
		}
	}
}
