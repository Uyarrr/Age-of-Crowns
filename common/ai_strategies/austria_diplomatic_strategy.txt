ai_strategy_hre_imperial_hegemony = {
	icon = "gfx/interface/icons/ai_strategy_icons/maintain_power_balance.dds"

	type = diplomatic

	# Austria already leads the Holy Roman Empire bloc.
	# This strategy maintains and expands it rather than creating another bloc.
	will_form_power_bloc = {
		always = no
	}

	undesirable_infamy_level = { value = 25 }
	unacceptable_infamy_level = { value = 50 }

	ideological_opinion_effect_mult = { value = 0.75 }

	wargoal_maneuvers_fraction = { value = 0.10 }

	# Austria should generally be willing to participate in relevant European plays.
	diplomatic_play_neutrality = { value = 0 }

	diplomatic_play_boldness = { value = 40 }

	recklessness = { value = 0.50 }


	# ---------------------------------------------------------------------
	# DEFEND THE HOLY ROMAN EMPIRE
	# ---------------------------------------------------------------------
	#
	# root                = Austria
	# scope:country       = country whose side Austria is evaluating
	# scope:enemy_country = leader of the opposing side
	# scope:is_initiator  = true when evaluating the initiator side
	#
	# HRE members attacked by outside powers receive overwhelming support.
	# Relations do NOT matter: imperial membership itself is the guarantee.
	#

	diplomatic_play_support = {
		value = 0


		# -------------------------------------------------------------
		# DEFEND HRE MEMBER AGAINST OUTSIDER
		# -------------------------------------------------------------

		if = {
			limit = {
				# We are evaluating the target/defending side.
				NOT = { exists = scope:is_initiator }

				scope:country = {
					is_in_same_power_bloc = root
				}

				scope:enemy_country = {
					NOT = {
						is_in_same_power_bloc = root
					}
				}
			}

			add = 1000
		}


		# -------------------------------------------------------------
		# NEVER SUPPORT AN OUTSIDER ATTACKING AN HRE MEMBER
		# -------------------------------------------------------------

		if = {
			limit = {
				# We are evaluating the initiating/attacking side.
				exists = scope:is_initiator

				scope:country = {
					NOT = {
						is_in_same_power_bloc = root
					}
				}

				scope:enemy_country = {
					is_in_same_power_bloc = root
				}
			}

			add = -1000
		}


		# -------------------------------------------------------------
		# INTERNAL HRE WARS
		# -------------------------------------------------------------
		#
		# Austria should normally remain neutral when both sides
		# are members of the Empire.
		#

		if = {
			limit = {
				scope:country = {
					is_in_same_power_bloc = root
				}

				scope:enemy_country = {
					is_in_same_power_bloc = root
				}
			}

			add = -500
		}
	}


	# ---------------------------------------------------------------------
	# AGGRESSION
	# ---------------------------------------------------------------------
	#
	# Austria should preserve the imperial system rather than conquer
	# members of the Empire.
	#

	aggression = {
		value = 0.25


		# No aggression against fellow HRE members.
		if = {
			limit = {
				is_in_same_power_bloc = scope:target_country
			}

			multiply = 0
		}


		# Subjects should be managed diplomatically rather than invaded.
		if = {
			limit = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			multiply = 0.05
		}


		# Modest expansion pressure against nearby weak,
		# independent and unaligned countries.
		if = {
			limit = {
				has_strategic_adjacency = scope:target_country

				scope:target_country.country_rank <= rank_value:minor_power

				scope:target_country = {
					is_subject = no
					is_in_power_bloc = no
				}
			}

			add = 0.35
		}


		# Ideological hostility has some influence,
		# but Austria remains relatively restrained.
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.05
			add = 1

			min = 0.75
			max = 1.25
		}

		max = 1.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOALS
	# ---------------------------------------------------------------------

	secret_goal_weights = {
		protect = 4.0
		befriend = 3.0
		reconcile = 2.0

		dominate = 1.25

		conquer = 0.10
	}


	# ---------------------------------------------------------------------
	# DIPLOMATIC / TREATY PRIORITIES
	# ---------------------------------------------------------------------
	#
	# Austria should build a cohesive diplomatic Empire rather than
	# directly annexing everything.
	#

	treaty_category_scores = {
		economy = { value = 4 }
		trade = { value = 5 }

		military = { value = 4 }
		military_defense = { value = 10 }

		ideology = { value = 1 }

		expansion = { value = 3 }

		power_bloc = { value = 10 }

		other = { value = 1 }
		none = { value = 0 }
	}


	# ---------------------------------------------------------------------
	# WARGOAL PRIORITIES
	# ---------------------------------------------------------------------

	wargoal_weights = {
		join_power_bloc = 5.0

		make_protectorate = 1.0
		make_dominion = 0.25

		reduce_autonomy = 1.5
		annex_country = 0.50

		liberate_subject = 1.5
		liberate_country = 1.25

		return_state = 0.75

		conquer_state = 0.10

		force_nationalization = 0.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL SCORING
	# ---------------------------------------------------------------------

	secret_goal_scores = {


		# -------------------------------------------------------------
		# PROTECT HRE MEMBERS
		# -------------------------------------------------------------

		protect = {
			trigger = {
				is_in_same_power_bloc = scope:target_country
			}

			score = {
				value = 600


				# Existing Austrian subjects are especially important.
				if = {
					limit = {
						scope:target_country = {
							is_subject_of = root
						}
					}

					add = 100
				}


				# Friendly members receive an additional preference,
				# but friendship is no longer required for protection.
				if = {
					limit = {
						scope:target_country = {
							relations:root >= relations_threshold:cordial
						}
					}

					add = 100
				}


				if = {
					limit = {
						scope:target_country = {
							relations:root >= relations_threshold:amicable
						}
					}

					add = 100
				}
			}
		}


		# -------------------------------------------------------------
		# BEFRIEND MEMBERS AND POTENTIAL MEMBERS
		# -------------------------------------------------------------

		befriend = {
			trigger = {
				OR = {

					# Existing HRE members.
					is_in_same_power_bloc = scope:target_country


					# Nearby potential imperial members.
					AND = {
						has_strategic_adjacency = scope:target_country

						scope:target_country.country_rank <= rank_value:minor_power

						scope:target_country = {
							is_subject = no
							is_in_power_bloc = no

							any_scope_state = {
								OR = {
									is_in_geographic_region = geographic_region_western_europe
									is_in_geographic_region = geographic_region_central_europe
									is_in_geographic_region = geographic_region_southern_europe
								}
							}
						}
					}
				}
			}

			score = {
				value = 100


				# Existing members are high diplomatic priorities.
				if = {
					limit = {
						is_in_same_power_bloc = scope:target_country
					}

					add = 350


					if = {
						limit = {
							scope:target_country = {
								relations:root < relations_threshold:cordial
							}
						}

						add = 300
					}


					if = {
						limit = {
							scope:target_country = {
								relations:root < relations_threshold:friendly
							}
						}

						add = 100
					}
				}


				# Potential members should be diplomatically cultivated.
				if = {
					limit = {
						NOT = {
							is_in_same_power_bloc = scope:target_country
						}
					}

					add = 250


					if = {
						limit = {
							scope:target_country = {
								relations:root < relations_threshold:amicable
							}
						}

						add = 200
					}
				}
			}
		}


		# -------------------------------------------------------------
		# RECONCILE WITH UNFRIENDLY MEMBERS
		# -------------------------------------------------------------

		reconcile = {
			trigger = {
				is_in_same_power_bloc = scope:target_country

				scope:target_country = {
					relations:root < relations_threshold:cordial
				}
			}

			score = {
				value = 500
			}
		}


		# -------------------------------------------------------------
		# DOMINATE POTENTIAL MEMBERS
		# -------------------------------------------------------------
		#
		# Coercion is a fallback when nearby candidates remain
		# outside the Empire and cannot be integrated peacefully.
		#

		dominate = {
			trigger = {
				has_strategic_adjacency = scope:target_country

				scope:target_country.country_rank <= rank_value:minor_power

				scope:target_country = {
					is_subject = no
					is_in_power_bloc = no

					relations:root < relations_threshold:cordial

					any_scope_state = {
						OR = {
							is_in_geographic_region = geographic_region_western_europe
							is_in_geographic_region = geographic_region_central_europe
							is_in_geographic_region = geographic_region_southern_europe
						}
					}
				}
			}

			score = {
				value = 150


				if = {
					limit = {
						scope:target_country = {
							num_allies < 1
						}
					}

					add = 75
				}


				if = {
					limit = {
						scope:target_country = {
							is_in_war = yes
						}
					}

					add = 50
				}
			}
		}


		# Avoid having no diplomatic intention toward unhappy HRE members.
		none = {
			trigger = {
				is_in_same_power_bloc = scope:target_country

				scope:target_country = {
					relations:root < relations_threshold:cordial
				}
			}

			score = {
				value = 25
			}
		}
	}


	# ---------------------------------------------------------------------
	# WARGOAL SCORING
	# ---------------------------------------------------------------------

	wargoal_scores = {


		# -------------------------------------------------------------
		# FORCE ENTRY INTO THE EMPIRE
		# -------------------------------------------------------------

		join_power_bloc = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}

					scope:target_country = {
						is_subject = no
						is_in_power_bloc = no
					}
				}

				add = 500
			}
		}


		# -------------------------------------------------------------
		# PROTECTORATE FALLBACK
		# -------------------------------------------------------------

		make_protectorate = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}
				}

				add = 100
			}
		}


		# -------------------------------------------------------------
		# SUBJECT MANAGEMENT
		# -------------------------------------------------------------

		reduce_autonomy = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				add = 100
			}
		}


		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}

					scope:target_country.country_rank <= rank_value:minor_power
				}

				add = 50
			}
		}
	}


	# ---------------------------------------------------------------------
	# POSSIBLE
	# ---------------------------------------------------------------------

	possible = {
		is_subject = no

		c:AUS ?= this

		is_power_bloc_leader = yes

		power_bloc ?= {
			has_identity = identity:identity_holy_roman_empire
		}
	}


	# ---------------------------------------------------------------------
	# STRATEGY SELECTION
	# ---------------------------------------------------------------------

	weight = {
		value = 1000
	}
}
