ai_strategy_france_royal_centralization = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = yes
	}

	undesirable_infamy_level = {
		value = 45
	}

	unacceptable_infamy_level = {
		value = 80
	}

	ideological_opinion_effect_mult = {
		value = 0.9
	}

	wargoal_maneuvers_fraction = {
		value = 0.25
	}

	diplomatic_play_neutrality = {
		value = 5
	}

	diplomatic_play_boldness = {
		value = 50
	}

	recklessness = {
		value = 1
	}

	aggression = {
		value = 1.15

		if = {
			limit = {
				has_technology_researched = colonization
				has_technology_researched = colonial_trade_networks
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				NOT = { scope:target_country = { is_subject_of = root } }
				scope:target_country = {
					any_scope_state = { region = sr:region_canada }
				}
			}
			add = 1.25
		}

		if = {
			limit = {
				scope:target_country = { c:ENG ?= this }
			}
			add = 2.5
		}

		if = {
			limit = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				has_strategic_adjacency = scope:target_country
				scope:target_country = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_france
					}
				}
			}
			add = 1.25
		}

		if = {
			limit = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				has_strategic_adjacency = scope:target_country
			}
			add = 0.6
		}

		if = {
			limit = {
				scope:target_country.country_rank < root.country_rank
			}
			add = 0.4
		}

		if = {
			limit = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				NOT = { has_strategic_adjacency = scope:target_country }
				scope:target_country = {
					NOT = {
						OR = {
							c:ENG ?= this
							any_scope_state = {
								is_in_geographic_region = geographic_region_france
							}
						}
					}
				}
			}
			add = -1.0
		}

		if = {
			limit = {
				OR = {
					scope:target_country = { is_subject_of = root }
					NOT = { aoc_france_diplomatic_priority = { TARGET = scope:target_country } }
				}
			}
			add = -10
		}
	}

	building_group_weights = {
		bg_army = 1.75
		bg_navy = 1.15
	}

	# France joins after developing a mature Era III colonial program. Later
	# priorities unlock from real colonial footholds rather than calendar dates.
	colonial_interest_ratio = {
		value = 0
		if = {
			limit = {
				has_technology_researched = colonization
				has_technology_researched = colonial_trade_networks
			}
			add = 0.55
		}
	}

	strategic_region_scores = {
		if = {
			limit = {
				scope:region = { is_in_geographic_region = geographic_region:geographic_region_europe }
			}
			add = 350
		}

		if = {
			limit = {
				scope:region = sr:region_west_africa
				NOT = {
					any_scope_state = { region = scope:region }
					any_direct_subject = { any_scope_state = { region = scope:region } }
				}
			}
			add = -1000
		}

		if = {
			limit = {
				has_technology_researched = colonization
				has_technology_researched = colonial_trade_networks
				scope:region = sr:region_canada
			}
			add = 850
		}

		if = {
			limit = {
				has_variable = aoc_french_atlantic_phase
				has_technology_researched = colonization
				scope:region = sr:region_atlantic_coast
			}
			add = 300
		}

		if = {
			limit = {
				has_variable = aoc_french_caribbean_phase
				has_technology_researched = colonization
				scope:region = sr:region_central_america
			}
			add = 200
		}
	}

	state_value = {
		value = 0
		if = {
			limit = {
				has_technology_researched = colonization
				has_technology_researched = colonial_trade_networks
				can_reach_target_state = { TARGET_STATE = scope:target_state }
				scope:target_state = { region = sr:region_canada }
			}
			add = 175
		}
		if = {
			limit = {
				has_variable = aoc_french_atlantic_phase
				has_technology_researched = colonization
				can_reach_target_state = { TARGET_STATE = scope:target_state }
				scope:target_state = { region = sr:region_atlantic_coast }
			}
			add = 75
		}
		min = 0
		max = 250
	}

	secret_goal_weights = {
		conquer = 2.5
		dominate = 0.9
		protect = 1.0
		befriend = 1.0
	}

	wargoal_weights = {
		conquer_state = 3.5
		return_state = 2.5
		make_tributary = 1.1
		make_dominion = 0.5
		reduce_autonomy = 0.0
		annex_country = 0.0
		force_nationalization = 0.0
	}

	secret_goal_scores = {
		none = {
			trigger = {
				NOT = { aoc_france_diplomatic_priority = { TARGET = scope:target_country } }
			}
			score = { value = 1000 }
		}

		conquer = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				OR = {
					scope:target_country = { c:ENG ?= this }
					has_strategic_adjacency = scope:target_country
					scope:target_country = {
						any_scope_state = {
							is_in_geographic_region = geographic_region_france
						}
					}
				}
			}
			score = {
				value = 120

				if = {
					limit = {
						scope:target_country = { c:ENG ?= this }
					}
					add = 450
				}

				if = {
					limit = {
						scope:target_country = {
							any_scope_state = {
								is_in_geographic_region = geographic_region_france
							}
						}
					}
					add = 200
				}

				if = {
					limit = {
						has_strategic_adjacency = scope:target_country
					}
					add = 100
				}

				if = {
					limit = {
						scope:target_country.country_rank <= rank_value:minor_power
					}
					add = 50
				}
			}
		}

		dominate = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				has_strategic_adjacency = scope:target_country
				scope:target_country.country_rank <= rank_value:minor_power
				scope:target_country = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_france
					}
				}
			}
			score = {
				value = 80
			}
		}

		protect = {
			trigger = {
				scope:target_country = {
					is_subject_of = root
					"relations(root)" >= 0
				}
			}
			score = {
				value = 350
			}
		}

		befriend = {
			trigger = {
				NOT = { scope:target_country = { c:ENG ?= this } }
				OR = {
					scope:target_country = { is_subject_of = root }
					AND = {
						c:FRA ?= this
						NOT = { has_global_variable = aoc_hyw_concluded }
						scope:target_country = {
							is_subject = no
							capital = { is_in_geographic_region = geographic_region:geographic_region_europe }
							has_strategic_region_interest_tier = {
								strategic_region = root.capital.region
								value >= 1
							}
							aoc_hyw_not_fighting_crown = { CROWN = c:FRA }
							NOT = { aoc_hyw_english_supporter = yes }
						}
					}
				}
			}
			score = {
				value = 200
				if = {
					limit = { scope:target_country = { is_subject_of = root } }
					add = 500
				}
			}
		}
	}

	wargoal_scores = {
		conquer_state = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = conquer
					}
				}
				add = 250
			}

			if = {
				limit = {
					scope:target_country = { c:ENG ?= this }
				}
				add = 300
			}

			if = {
				limit = {
					scope:target_state = {
						is_in_geographic_region = geographic_region_france
					}
				}
				add = 125
			}

			if = {
				limit = {
					scope:target_state = {
						any_neighbouring_state = {
							owner = root
						}
					}
				}
				add = 100
			}
		}

		return_state = {
			if = {
				limit = {
					scope:target_country = { c:ENG ?= this }
				}
				add = 300
			}

			if = {
				limit = {
					scope:target_state = {
						is_in_geographic_region = geographic_region_france
					}
				}
				add = 150
			}
		}

		make_tributary = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}
					scope:target_country.country_rank <= rank_value:minor_power
				}
				add = 125
			}
		}

		make_dominion = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}
					scope:target_country.country_rank <= rank_value:minor_power
				}
				add = 50
			}
		}

		reduce_autonomy = {
			if = {
				limit = { scope:target_country = { is_subject_of = root } }
				add = -1000
			}
		}

		annex_country = {
			if = {
				limit = { scope:target_country = { is_subject_of = root } }
				add = -1000
			}
		}
	}

	possible = {
		is_subject = no
		OR = {
			c:FRA ?= this
			was_formed_from = FRA
		}
	}

	weight = {
		value = 150

		if = {
			limit = {
				exists = c:ENG
			}
			add = 100
		}

		if = {
			limit = {
				c:ENG = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_france
					}
				}
			}
			add = 150
		}

		if = {
			limit = {
				any_country = {
					is_subject_of = root
				}
			}
			add = 75
		}
	}
}
