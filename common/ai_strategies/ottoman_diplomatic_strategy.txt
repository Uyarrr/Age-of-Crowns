ai_strategy_ottoman_imperial_expansion = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = yes
	}

	undesirable_infamy_level = { value = 65 }
	unacceptable_infamy_level = { value = 95 }

	ideological_opinion_effect_mult = { value = 0.75 }

	wargoal_maneuvers_fraction = { value = 0.35 }

	diplomatic_play_neutrality = { value = 5 }
	diplomatic_play_boldness = { value = 65 }

	recklessness = { value = 1.35 }


	# ---------------------------------------------------------------------
	# AGGRESSION
	# ---------------------------------------------------------------------
	#
	# Ottoman expansion follows the frontier.
	#
	# PRIORITY:
	# 1. Byzantium
	# 2. Claimed frontier territory
	# 3. Frontier countries in Ottoman expansion regions
	# 4. Weaker frontier powers
	#
	# Distant countries should generally be ignored even when technically
	# reachable through naval or diplomatic-play mechanics.
	#

	aggression = {
		value = 0.15


		# -------------------------------------------------------------
		# BYZANTIUM
		# -------------------------------------------------------------
		#
		# Byzantium is the major historical / strategic exception.
		#

		if = {
			limit = {
				scope:target_country = {
					c:BYZ ?= this
				}

				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}
			}

			add = 6.0
		}


		# -------------------------------------------------------------
		# CLAIMED FRONTIER TERRITORY
		# -------------------------------------------------------------
		#
		# Claims matter strongly ONLY when they are actually on the
		# Ottoman frontier.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				scope:target_country = {
					any_scope_state = {
						has_claim_by = root

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 3.0
		}


		# -------------------------------------------------------------
		# REGIONAL FRONTIER EXPANSION
		# -------------------------------------------------------------
		#
		# Being in one of these regions is NOT enough.
		# The target must actually border Ottoman territory or an
		# Ottoman subject.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				scope:target_country = {
					any_scope_state = {

						OR = {
							is_in_geographic_region = geographic_region_balkans
							is_in_geographic_region = geographic_region_near_east
							is_in_geographic_region = geographic_region_arabia
							is_in_geographic_region = geographic_region_nile_basin
							is_in_geographic_region = geographic_region_north_africa
							is_in_geographic_region = geographic_region_greater_persia
						}

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 1.75
		}


		# -------------------------------------------------------------
		# WEAKER FRONTIER POWERS
		# -------------------------------------------------------------

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				scope:target_country.country_rank < root.country_rank

				scope:target_country = {
					any_scope_state = {
						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 0.50
		}


		# -------------------------------------------------------------
		# OPPORTUNISM
		# -------------------------------------------------------------
		#
		# Only relevant against countries already on the frontier.
		#

		if = {
			limit = {
				scope:target_country = {
					is_in_war = yes

					any_scope_state = {
						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 0.25
		}


		if = {
			limit = {
				scope:target_country = {
					num_allies < 1

					any_scope_state = {
						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 0.20
		}


		# -------------------------------------------------------------
		# SUBJECTS
		# -------------------------------------------------------------

		if = {
			limit = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			multiply = 0.05
		}


		# -------------------------------------------------------------
		# IRRELEVANT / DISTANT TARGETS
		# -------------------------------------------------------------
		#
		# If the target is:
		# - not Byzantium
		# - not holding a claimed frontier state
		# - not holding a desired regional frontier state
		#
		# then Ottoman aggression toward it should be negligible.
		#

		if = {
			limit = {
				NOT = {
					scope:target_country = {
						c:BYZ ?= this
					}
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							has_claim_by = root

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							OR = {
								is_in_geographic_region = geographic_region_balkans
								is_in_geographic_region = geographic_region_near_east
								is_in_geographic_region = geographic_region_arabia
								is_in_geographic_region = geographic_region_nile_basin
								is_in_geographic_region = geographic_region_north_africa
								is_in_geographic_region = geographic_region_greater_persia
							}

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}
			}

			multiply = 0.05
		}


		max = 7.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL WEIGHTS
	# ---------------------------------------------------------------------

	secret_goal_weights = {
		conquer = 3.5
		dominate = 0.65

		protect = 0.25

		befriend = 0.0
		reconcile = 0.0
	}


	# ---------------------------------------------------------------------
	# WARGOAL WEIGHTS
	# ---------------------------------------------------------------------

	wargoal_weights = {
		conquer_state = 4.0
		return_state = 3.0

		make_protectorate = 0.65
		make_dominion = 0.25

		reduce_autonomy = 1.25
		annex_country = 1.75

		force_nationalization = 0.0
	}


	# ---------------------------------------------------------------------
	# SECRET GOAL SCORING
	# ---------------------------------------------------------------------

	secret_goal_scores = {


		# -------------------------------------------------------------
		# CONQUER
		# -------------------------------------------------------------

		conquer = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				OR = {

					# Byzantium is always strategically important.
					scope:target_country = {
						c:BYZ ?= this
					}


					# Claimed frontier territory.
					scope:target_country = {
						any_scope_state = {
							has_claim_by = root

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}


					# Regional frontier expansion.
					scope:target_country = {
						any_scope_state = {
							OR = {
								is_in_geographic_region = geographic_region_balkans
								is_in_geographic_region = geographic_region_near_east
								is_in_geographic_region = geographic_region_arabia
								is_in_geographic_region = geographic_region_nile_basin
								is_in_geographic_region = geographic_region_north_africa
								is_in_geographic_region = geographic_region_greater_persia
							}

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}
			}

			score = {
				value = 100


				# Byzantium overwhelmingly takes priority.
				if = {
					limit = {
						scope:target_country = {
							c:BYZ ?= this
						}
					}

					add = 1500
				}


				# Claimed frontier states.
				if = {
					limit = {
						scope:target_country = {
							any_scope_state = {
								has_claim_by = root

								any_neighbouring_state = {
									OR = {
										owner = root

										owner = {
											is_subject_of = root
										}
									}
								}
							}
						}
					}

					add = 600
				}


				# Desired-region frontier targets.
				if = {
					limit = {
						scope:target_country = {
							any_scope_state = {
								OR = {
									is_in_geographic_region = geographic_region_balkans
									is_in_geographic_region = geographic_region_near_east
									is_in_geographic_region = geographic_region_arabia
									is_in_geographic_region = geographic_region_nile_basin
									is_in_geographic_region = geographic_region_north_africa
									is_in_geographic_region = geographic_region_greater_persia
								}

								any_neighbouring_state = {
									OR = {
										owner = root

										owner = {
											is_subject_of = root
										}
									}
								}
							}
						}
					}

					add = 300
				}


				# Prefer weaker targets.
				if = {
					limit = {
						scope:target_country.country_rank < root.country_rank
					}

					add = 100
				}


				# Opportunistic wars.
				if = {
					limit = {
						scope:target_country = {
							is_in_war = yes
						}
					}

					add = 75
				}


				if = {
					limit = {
						scope:target_country = {
							num_allies < 1
						}
					}

					add = 50
				}
			}
		}


		# -------------------------------------------------------------
		# DOMINATE
		# -------------------------------------------------------------
		#
		# Small regional frontier countries can be made subjects instead
		# of conquered directly.
		#
		# Claimed land and Byzantium should be conquered instead.
		#

		dominate = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				scope:target_country.country_rank <= rank_value:minor_power

				NOT = {
					scope:target_country = {
						c:BYZ ?= this
					}
				}

				scope:target_country = {
					any_scope_state = {
						OR = {
							is_in_geographic_region = geographic_region_balkans
							is_in_geographic_region = geographic_region_near_east
							is_in_geographic_region = geographic_region_arabia
							is_in_geographic_region = geographic_region_nile_basin
							is_in_geographic_region = geographic_region_north_africa
							is_in_geographic_region = geographic_region_greater_persia
						}

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}

					NOT = {
						any_scope_state = {
							has_claim_by = root
						}
					}
				}
			}

			score = {
				value = 100


				if = {
					limit = {
						scope:target_country = {
							num_allies < 1
						}
					}

					add = 75
				}


				if = {
					limit = {
						scope:target_country = {
							is_in_war = yes
						}
					}

					add = 50
				}
			}
		}


		# -------------------------------------------------------------
		# PROTECT SUBJECTS
		# -------------------------------------------------------------

		protect = {
			trigger = {
				scope:target_country = {
					is_subject_of = root
				}
			}

			score = {
				value = 125
			}
		}


		# -------------------------------------------------------------
		# IGNORE DISTANT / IRRELEVANT COUNTRIES
		# -------------------------------------------------------------

		none = {
			trigger = {
				NOT = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				NOT = {
					scope:target_country = {
						c:BYZ ?= this
					}
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							has_claim_by = root

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}

				NOT = {
					scope:target_country = {
						any_scope_state = {
							OR = {
								is_in_geographic_region = geographic_region_balkans
								is_in_geographic_region = geographic_region_near_east
								is_in_geographic_region = geographic_region_arabia
								is_in_geographic_region = geographic_region_nile_basin
								is_in_geographic_region = geographic_region_north_africa
								is_in_geographic_region = geographic_region_greater_persia
							}

							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}
			}

			score = {
				value = 1000
			}
		}
	}


	# ---------------------------------------------------------------------
	# WARGOAL SCORING
	# ---------------------------------------------------------------------

	wargoal_scores = {


		# -------------------------------------------------------------
		# CONQUER STATE
		# -------------------------------------------------------------

		conquer_state = {

			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = conquer
					}
				}

				add = 250
			}


			# Byzantium remains the highest priority.
			if = {
				limit = {
					scope:target_country = {
						c:BYZ ?= this
					}
				}

				add = 750
			}


			# Claims matter strongly.
			if = {
				limit = {
					scope:target_state = {
						has_claim_by = root
					}
				}

				add = 500
			}


			# Contiguous states are strongly preferred.
			if = {
				limit = {
					scope:target_state = {
						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}

				add = 300
			}


			# Desired expansion regions only receive a bonus when
			# the target state is actually on the frontier.
			if = {
				limit = {
					scope:target_state = {
						OR = {
							is_in_geographic_region = geographic_region_balkans
							is_in_geographic_region = geographic_region_near_east
							is_in_geographic_region = geographic_region_arabia
							is_in_geographic_region = geographic_region_nile_basin
							is_in_geographic_region = geographic_region_north_africa
							is_in_geographic_region = geographic_region_greater_persia
						}

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}

				add = 150
			}


			# Strongly punish disconnected conquest.
			if = {
				limit = {
					NOT = {
						scope:target_country = {
							c:BYZ ?= this
						}
					}

					NOT = {
						scope:target_state = {
							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}

				add = -600
			}
		}


		# -------------------------------------------------------------
		# RETURN STATE
		# -------------------------------------------------------------

		return_state = {

			if = {
				limit = {
					scope:target_state = {
						has_claim_by = root
					}
				}

				add = 400
			}


			if = {
				limit = {
					scope:target_state = {
						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}

				add = 200
			}


			if = {
				limit = {
					NOT = {
						scope:target_state = {
							any_neighbouring_state = {
								OR = {
									owner = root

									owner = {
										is_subject_of = root
									}
								}
							}
						}
					}
				}

				add = -300
			}
		}


		# -------------------------------------------------------------
		# PROTECTORATE
		# -------------------------------------------------------------

		make_protectorate = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}

					scope:target_country.country_rank <= rank_value:minor_power
				}

				add = 100
			}
		}


		# -------------------------------------------------------------
		# DOMINION
		# -------------------------------------------------------------

		make_dominion = {
			if = {
				limit = {
					has_secret_goal = {
						who = scope:target_country
						secret_goal = dominate
					}

					scope:target_country.country_rank <= rank_value:minor_power
				}

				add = 25
			}
		}


		# -------------------------------------------------------------
		# SUBJECT MANAGEMENT
		# -------------------------------------------------------------

		reduce_autonomy = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}
				}

				add = 125
			}
		}


		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
					}

					scope:target_country.country_rank <= rank_value:minor_power
				}

				add = 175
			}
		}
	}


	# ---------------------------------------------------------------------
	# STATE VALUE
	# ---------------------------------------------------------------------
	#
	# State value also follows the Ottoman frontier.
	#

	state_value = {
		value = 0


		# Ottoman-owned states retain normal strategic relevance.
		if = {
			limit = {
				scope:target_state = {
					owner = root
				}
			}

			add = 25
		}


		# Claims.
		if = {
			limit = {
				scope:target_state = {
					has_claim_by = root
				}
			}

			add = 100
		}


		# Byzantium.
		if = {
			limit = {
				scope:target_country = {
					c:BYZ ?= this
				}
			}

			add = 150
		}


		# Border states.
		if = {
			limit = {
				scope:target_state = {
					any_neighbouring_state = {
						OR = {
							owner = root

							owner = {
								is_subject_of = root
							}
						}
					}
				}
			}

			add = 75
		}


		# Desired regions receive additional value ONLY when contiguous.
		if = {
			limit = {
				scope:target_state = {
					OR = {
						is_in_geographic_region = geographic_region_balkans
						is_in_geographic_region = geographic_region_near_east
						is_in_geographic_region = geographic_region_arabia
						is_in_geographic_region = geographic_region_nile_basin
						is_in_geographic_region = geographic_region_north_africa
						is_in_geographic_region = geographic_region_greater_persia
					}

					any_neighbouring_state = {
						OR = {
							owner = root

							owner = {
								is_subject_of = root
							}
						}
					}
				}
			}

			add = 50
		}


		min = 0
		max = 250
	}


	# ---------------------------------------------------------------------
	# AVAILABILITY
	# ---------------------------------------------------------------------

	possible = {
		is_subject = no

		c:TUR ?= this
	}


	# ---------------------------------------------------------------------
	# STRATEGY WEIGHT
	# ---------------------------------------------------------------------

	weight = {
		value = 100


		# Early Ottoman expansion period.
		if = {
			limit = {
				game_date < 1475.1.1
			}

			add = 200
		}


		# Byzantium existing independently makes this strategy
		# overwhelmingly important.
		if = {
			limit = {
				exists = c:BYZ

				c:BYZ = {
					NOT = {
						is_subject_of = root
					}
				}
			}

			add = 400
		}


		# Claimed frontier territory.
		if = {
			limit = {
				any_country = {
					NOT = {
						this = root
					}

					NOT = {
						is_subject_of = root
					}

					any_scope_state = {
						has_claim_by = root

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 150
		}


		# There are still valid regional frontier expansion targets.
		if = {
			limit = {
				any_country = {
					NOT = {
						this = root
					}

					NOT = {
						is_subject_of = root
					}

					any_scope_state = {
						OR = {
							is_in_geographic_region = geographic_region_balkans
							is_in_geographic_region = geographic_region_near_east
							is_in_geographic_region = geographic_region_arabia
							is_in_geographic_region = geographic_region_nile_basin
							is_in_geographic_region = geographic_region_north_africa
							is_in_geographic_region = geographic_region_greater_persia
						}

						any_neighbouring_state = {
							OR = {
								owner = root

								owner = {
									is_subject_of = root
								}
							}
						}
					}
				}
			}

			add = 150
		}
	}
}
