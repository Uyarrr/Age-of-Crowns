# Age of Crowns - generic diplomatic strategies
# Used by every independent country without a country-specific strategy.
# Country strategies (england_, france_, ...) use weights of 100+ so they win wherever they are possible.
# Names match vanilla so vanilla diplomatic actions, catalysts and decisions that check has_strategy keep working.

ai_strategy_territorial_expansion = {
	icon = "gfx/interface/icons/ai_strategy_icons/territorial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		global_country_ranking <= 3
	}

	undesirable_infamy_level = {
		value = 50
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = {
		value = 0.75
	}

	wargoal_maneuvers_fraction = {
		value = 0.15
	}

	diplomatic_play_neutrality = {
		value = 25
	}

	diplomatic_play_boldness = {
		value = 25
	}

	recklessness = {
		value = 0.25
	}

	aggression = {
		value = 1

		if = {
			limit = {
				NOT = { is_country_type = unrecognized }
				scope:target_country = { is_country_type = unrecognized }
			}
			add = 1
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_army = 1.5
	}

	secret_goal_weights = {
		protect = 0.5
		conquer = 1.5
	}

	wargoal_weights = {
		annex_country = 1.5
		conquer_state = 1.5
		return_state = 1.5
		force_nationalization = 0.25
	}

	secret_goal_scores = {
		conquer = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				can_reach_target_country = { TARGET_COUNTRY = scope:target_country }
				has_strategic_adjacency = scope:target_country
				scope:target_country.country_rank <= root.country_rank
			}
			score = {
				if = {
					limit = {
						NOT = { is_country_type = unrecognized }
						scope:target_country = { NOT = { is_country_type = unrecognized } }
					}
					add = 25
				}
				else_if = {
					limit = {
						scope:target_country = { is_country_type = unrecognized }
					}
					add = 50
				}
			}
		}
	}

	diplomatic_play_support = {
		value = 0

		if = { # Weakening potential conquest target
			limit = {
				scope:enemy_country = {
					has_strategic_adjacency = root
					any_scope_state = {
						OR = {
							is_homeland_of_country_cultures = root
							has_claim_by = root
						}
					}
				}
			}
			add = {
				value = 30
				desc = "AI_STRATEGY_SUPPORT_EXPANSION_WEAKEN_NEIGHBOUR"
			}
		}

		if = { # Little interest in other wars
			limit = {
				NOT = { has_strategic_adjacency = scope:country }
				NOT = { has_strategic_adjacency = scope:enemy_country }
			}
			add = {
				value = -20
				desc = "AI_STRATEGY_SUPPORT_EXPANSION_DISTANT_QUARREL"
			}
		}
	}

	treaty_category_scores = {
		expansion = {
			value = 4
		}

		military = {
			value = 3
		}
	}

	strategic_region_stance_type_scores = {
		stance_conquer_region = {
			value = 25
		}
	}

	possible = {
		is_subject = no
	}

	weight = {
		value = 10

		# AI implications for Power Blocs: 'Territorial Expansion' strategy is more likely for Blocs with Aggressive Coordination
		if = {
			limit = {
				power_bloc ?= {
					has_principle_group = principle_group:principle_group_aggressive_coordination
				}
			}
			add = 5
		}

		# High-fervour AI is inclined to take its homelands
		if = {
			limit = {
				country_fervor_primary_culture > 10
			}
			add = {
				value = country_fervor_primary_culture
				divide = 2
			}
		}

		if = {
			limit = { ig:ig_armed_forces ?= { is_powerful = yes } }
			multiply = 1.5
		}
	}
}

ai_strategy_maintain_power_balance = {
	icon = "gfx/interface/icons/ai_strategy_icons/maintain_power_balance.dds"

	type = diplomatic

	will_form_power_bloc = {
		global_country_ranking = 1
	}

	undesirable_infamy_level = {
		value = 25
	}

	unacceptable_infamy_level = {
		value = 50
	}

	ideological_opinion_effect_mult = {
		value = 1.5
	}

	wargoal_maneuvers_fraction = {
		value = -0.1
	}

	diplomatic_play_neutrality = {
		value = 0
	}

	diplomatic_play_boldness = {
		value = -25
	}

	recklessness = {
		value = -0.25
	}

	aggression = {
		value = 0.25

		if = {
			limit = {
				NOT = { is_country_type = unrecognized }
				scope:target_country = { is_country_type = unrecognized }
				has_technology_researched = colonization
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 1
				}
			}
			add = 0.25

			if = {
				limit = {
					power_bloc ?= { has_principle = principle:principle_colonial_offices_3 }
				}
				add = 0.5
			}
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.1
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_technology = 1.25
	}

	secret_goal_weights = {
		protect = 1.5
		dominate = 1.5
		reconcile = 1.5
	}

	wargoal_weights = {
		regime_change = 3.0
		humiliation = 2.0
		liberate_subject = 1.5
		liberate_country = 1.5
		force_nationalization = 0.25
	}

	diplomatic_play_support = {
		value = 0

		if = { # More likely to side with weaker targets
			limit = {
				scope:enemy_country = { country_rank >= rank_value:great_power }
				scope:country.country_rank < scope:enemy_country.country_rank
			}
			add = {
				value = 10
				desc = "AI_STRATEGY_SUPPORT_BALANCE_WEAKER_SIDE"
			}
		}

		if = { # Fun police against high infamy targets
			limit = {
				scope:enemy_country = {
					country_rank >= rank_value:great_power
					infamy >= infamy_threshold:infamous
				}
			}
			add = {
				value = 25
				desc = "AI_STRATEGY_SUPPORT_BALANCE_RESTRAIN_INFAMOUS"
			}
		}

		if = { # No desire to help rival bloc
			limit = {
				country_is_in_different_power_bloc = { TARGET = scope:country }
			}
			add = {
				value = -15
				desc = "AI_STRATEGY_SUPPORT_BALANCE_RIVAL_BLOC"
			}
		}
	}

	treaty_category_scores = {
		military_defense = {
			value = 4
		}

		expansion = {
			value = -2
		}
	}

	strategic_region_stance_type_scores = {
		stance_protect_region = {
			value = 25
		}

		stance_conquer_region = {
			value = -25
		}
	}

	obligation_value = {
		if = {
			limit = {
				scope:target_country = { country_rank >= rank_value:major_power }
			}
			add = 5
		}
	}

	possible = {
		is_subject = no
	}

	weight = {
		value = 10

		if = {
			limit = { country_rank < rank_value:major_power }
			add = 10
		}
	}
}

ai_strategy_colonial_expansion = {
	icon = "gfx/interface/icons/ai_strategy_icons/colonial_expansion.dds"

	type = diplomatic

	will_form_power_bloc = {
		global_country_ranking <= 3
	}

	undesirable_infamy_level = {
		value = 25
	}

	unacceptable_infamy_level = {
		value = 50
	}

	ideological_opinion_effect_mult = {
		value = 1.0
	}

	wargoal_maneuvers_fraction = {
		value = 0

		if = {
			limit = {
				scope:enemy_country = { is_country_type = unrecognized }
			}
			add = 0.25
		}
	}

	diplomatic_play_neutrality = {
		value = 25
	}

	diplomatic_play_boldness = {
		value = -15
	}

	recklessness = {
		value = 0.1
	}

	# Added to value from default strategy
	aggression = {
		value = 0.25
		if = {
			limit = {
				NOT = { is_country_type = unrecognized }
				scope:target_country = { ai_country_is_valid_colonization_target = yes }
				has_technology_researched = colonization
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 1
				}
			}
			add = 0.25

			if = {
				limit = {
					power_bloc ?= { has_principle = principle:principle_colonial_offices_3 }
				}
				add = 0.75
			}
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.1
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_navy = 1.5
	}

	colonial_interest_ratio = {
		value = 0.10

		if = {
			limit = { navy_size >= 30 }
			add = 0.1
		}

		if = {
			limit = { navy_size >= 60 }
			add = 0.1
		}

		if = {
			limit = { navy_size >= 90 }
			add = 0.1
		}

		if = {
			limit = { navy_size >= 120 }
			add = 0.1
		}
	}

	strategic_region_scores = {
		if = {
			limit = {
				scope:region = {
					any_decentralized_country = {
						exists = this
					}
				}
			}
			add = 50
		}
	}

	secret_goal_scores = {
		dominate = {
			score = {
				if = {
					limit = {
						scope:target_country = { ai_country_is_valid_colonization_target = no }
					}
					add = -100
				}
				else_if = {
					limit = {
						ai_country_is_valid_colonization_target = yes
					}
					add = 200
				}
			}
		}
		conquer = {
			score = {
				if = {
					limit = {
						scope:target_country = { NOT = { is_country_type = unrecognized } }
					}
					add = -100
				}
				if = {
					limit = {
						NOT = { scope:target_country = { is_subject_of = root } }
						has_strategic_adjacency = scope:target_country
						scope:target_country = { is_country_type = unrecognized }
					}
					add = 200

					if = {
						limit = {
							scope:target_country = {
								is_subject = yes
								is_country_type = unrecognized
								top_overlord ?= {
									NOT = { is_country_type = unrecognized }
								}
							}
						}
						multiply = 0.0
					}
				}
			}
		}
	}

	wargoal_weights = {
		conquer_state = 1.5
		reduce_autonomy = 1.5
		take_treaty_port = 1.5
		force_nationalization = 0.5
	}

	diplomatic_play_support = {
		value = 0

		if = { # Maintain superiority over colonial regions by supporting locals
			limit = {
				has_overlapping_interests = scope:enemy_country
				scope:enemy_country = { country_rank >= rank_value:major_power }
			}
			add = {
				value = 20
				desc = "AI_STRATEGY_SUPPORT_COLONIAL_DENY_RIVAL"
			}
		}

		if = { # Focused on colonizing
			limit = {
				scope:country = {
					NOR = {
						is_country_type = unrecognized
						is_country_type = decentralized
					}
				}
				NOT = { has_overlapping_interests = scope:country }
			}
			add = {
				value = -20
				desc = "AI_STRATEGY_SUPPORT_COLONIAL_NOT_OUR_CONCERN"
			}
		}
	}

	treaty_category_scores = {
		expansion = {
			value = 3
		}

		trade = {
			value = 2
		}
	}

	strategic_region_stance_type_scores = {
		stance_colonize_region = {
			value = 30
		}
	}

	max_active_stances = {
		add = 2
	}

	obligation_value = {
		if = { # Obligations from fellow colonial powers keep the carve-up orderly
			limit = {
				scope:target_country = {
					country_rank >= rank_value:major_power
					NOT = { is_country_type = unrecognized }
				}
			}
			add = 5
		}
	}

	possible = {
		is_subject = no
		is_country_type = recognized
		OR = {
			has_law_or_variant = law_type:law_colonial_resettlement
			has_law_or_variant = law_type:law_colonial_exploitation
		}
	}

	weight = {
		value = 10

		if = {
			limit = { country_rank = rank_value:major_power }
			add = 5
		}

		if = {
			limit = { country_rank = rank_value:great_power }
			add = 5
		}

		# AI implications for Power Blocs: 'Colonial Expansion' strategy is more likely for Blocs with Colonial Offices
		if = {
			limit = {
				power_bloc ?= {
					has_principle_group = principle_group:principle_group_colonial_offices
				}
			}
			add = 5
		}

		if = {
			limit = { navy_size < 50 }
			multiply = 0.5
		}

		if = {
			limit = { navy_size >= 75 }
			multiply = 1.5
		}

		if = {
			limit = {
				power_bloc ?= { has_principle_group = principle_group:principle_group_colonial_offices }
			}
			multiply = 1.5
		}
	}
}
