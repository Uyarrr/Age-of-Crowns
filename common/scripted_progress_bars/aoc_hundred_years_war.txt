			limit = { aoc_hyw_round_open = no }
			add = { desc = aoc_hyw_tension_rivalry value = 1.5 }
		}
		progress_condition = {
			limit = {
				aoc_hyw_round_open = no
				c:ENG ?= { has_variable = aoc_hyw_maine_refused }

					any_scope_state = {
						owner = {
							OR = {
								c:ENG ?= this
								is_subject_of = c:ENG
							}
						}
					}
				}
			}
			add = { desc = aoc_hyw_tension_maine value = 0.5 }
		}
		progress_condition = {
			limit = {
				aoc_hyw_round_open = no
				c:ENG ?= { has_variable = aoc_hyw_claimed_throne }
			}
			add = { desc = aoc_hyw_tension_claim value = 1 }
		}
		progress_condition = {
			limit = { aoc_hyw_round_open = yes }
			add = { desc = aoc_hyw_tension_war value = -3 }
		}
	}
