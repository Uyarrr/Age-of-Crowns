# Courtiers and captains eager for conquest, crusade and overseas empire.
# The Crusade of Varna (1444), Charles the Bold, the completion of the
# Reconquista (1492), the Italian Wars (1494) and the conquistadors.
movement_war_party = {
	category = movement_category_ideological

	ideology = ideology_outward
	character_ideologies = {
		ideology_jingoist_leader
		ideology_authoritarian
	}

	creation_trigger = {}
	creation_weight = {
		value = 100
		if = {
			limit = { number_of_claims > 0 }
			multiply = 2
		}
		if = {
			limit = { has_technology_researched = colonization }
			multiply = 1.5
		}
		if = {
			limit = { has_law_or_variant = law_type:law_civil_society }
			multiply = 1.5
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_jingoist_leader
			scope:interest_group ?= {
				OR = {
					is_interest_group_type = ig_armed_forces
					has_ideology = ideology:ideology_outward
				}
			}
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = { has_ideology = ideology:ideology_jingoist_leader }
			multiply = 5
		}
	}

	pop_support_trigger = {
		NOT = { is_pop_type = slaves }
	}

	pop_support_factors = {
		movement_support_officers
		movement_support_aristocrats
		movement_support_soldiers
		movement_support_number_of_claims
		movement_support_colonial_settlers
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_suppressed = yes }
			}
			add = {
				desc = "concept_suppressed"
				value = suppression_base_value
				multiply = {
					value = -0.5
					add = owner.modifier:country_suppression_attraction_factor
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_bolstered = yes }
			}
			add = {
				desc = "concept_bolstered"
				value = bolstering_base_value
				multiply = {
					value = 0.5
					add = owner.modifier:country_bolster_attraction_factor
				}
			}
		}

		add = {
			value = 2
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = { is_pop_type = officers }
			add = {
				value = 40
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = { is_pop_type = aristocrats }
			add = {
				value = 20
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = { is_pop_type = soldiers }
			add = {
				value = 15
				desc = "POP_SOLDIERS"
			}
		}
		else_if = {
			limit = { is_pop_type = capitalists }
			add = {
				value = 5
				desc = "POP_CAPITALISTS"
			}
		}

		if = {
			limit = {
				owner = { number_of_claims > 0 }
			}
			multiply = {
				value = 1.25
				desc = "AOC_MOVEMENT_UNREDEEMED_CLAIMS"
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 50
		}

		state_weight = {
			value = 1
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { number_of_claims > 0 }
			}
			add = {
				value = 0.05
				desc = "AOC_MOVEMENT_UNREDEEMED_CLAIMS"
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}
