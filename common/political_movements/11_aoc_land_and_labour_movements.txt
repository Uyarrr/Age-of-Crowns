# AGE OF CROWNS - LAND AND LABOUR

# the land, the commons and the workshop.

# Landlords binding the peasantry to the estate: the "second serfdom".
# The Statute of Piotrkow (1496), the Russian Sudebnik (1497) and the
# Ulozhenie (1649), Brandenburg and Livonian Gutsherrschaft.
movement_manorial = {
	category = movement_category_ideological

	ideology = ideology_serfdom
	character_ideologies = {
		ideology_moderate
		ideology_protectionist
	}

	creation_trigger = {
		NOT = { has_law_or_variant = law_type:law_chiefdom }
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_tenant_farmers
					has_law_or_variant = law_type:law_homesteading
				}
			}
			multiply = 2
		}
		if = {
			limit = { has_law_or_variant = law_type:law_serfdom }
			multiply = 0.5
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		scope:interest_group ?= {
			OR = {
				is_interest_group_type = ig_landowners
				has_ideology = ideology:ideology_serfdom
			}
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				scope:interest_group ?= { has_ideology = ideology:ideology_serfdom }
			}
			multiply = 3
		}
	}

	pop_support_trigger = {
		NOT = { is_pop_type = slaves }
	}

	pop_support_factors = {
		movement_support_aristocrats
		movement_support_officers
		movement_support_clergymen
		movement_support_low_urbanization
		movement_support_rural
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_suppressed = yes }
			}
			add = {
				desc = "concept_suppressed"
				value = suppression_base_value
				multiply = {
					value = -0.5
					add = owner.modifier:country_suppression_attraction_factor
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_bolstered = yes }
			}
			add = {
				desc = "concept_bolstered"
				value = bolstering_base_value
				multiply = {
					value = 0.5
					add = owner.modifier:country_bolster_attraction_factor
				}
			}
		}

		add = {
			value = 1
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = { is_pop_type = aristocrats }
			add = {
				value = 45
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = { is_pop_type = officers }
			add = {
				value = 6
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = { is_pop_type = clergymen }
			add = {
				value = 5
				desc = "POP_CLERGYMEN"
			}
		}
		else_if = {
			limit = {
				OR = {
					is_pop_type = peasants
					is_pop_type = farmers
					is_pop_type = laborers
				}
			}
			add = {
				value = -1
				desc = "LOWER_NO_ICON"
			}
		}

		multiply = {
			desc = "URBAN_STATE"
			value = 1.5
			subtract = state.state_urbanization_rate
			min = 0.5
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			subtract = state_urbanization_rate
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { has_law_or_variant = law_type:law_homesteading }
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_PEASANT_FREEHOLDS"
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}

			}
		}
		else_if = {
			limit = { is_pop_type = capitalists }
			add = {
				value = -2
				desc = "POP_CAPITALISTS"
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 50
		}

		state_weight = {
			value = state_urbanization_rate
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { has_law_or_variant = law_type:law_economic_liberalism }
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_GUILD_PRIVILEGES"
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}
