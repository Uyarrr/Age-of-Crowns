# Partisans of undivided royal sovereignty.
# Louis XI after the League of the Public Weal, the Catholic Monarchs,
# Ivan III, and later Bourbon and Danish absolutism (Lex Regia, 1665).
movement_absolutist = {
	category = movement_category_ideological

	ideology = ideology_absolutism
	character_ideologies = {
		ideology_authoritarian
		ideology_moderate
	}

	creation_trigger = {
		aoc_country_is_monarchy = yes
		has_technology_researched = sovereignty
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_strengthen_the_nobility
					aoc_country_is_decentralized = yes
				}
			}
			multiply = 2
		}
		if = {
			limit = { has_technology_researched = absolutism }
			multiply = 2
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	disband_trigger = {
		NOT = { aoc_country_is_monarchy = yes }
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_authoritarian
			scope:interest_group ?= {
				OR = {
					is_interest_group_type = ig_crown
					has_ideology = ideology:ideology_absolutism
				}
			}
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = { has_ideology = ideology:ideology_authoritarian }
			multiply = 5
		}
		if = {
			limit = {
				scope:interest_group ?= { is_interest_group_type = ig_crown }
			}
			multiply = 3
		}
	}

	pop_support_trigger = {
		NOT = { is_pop_type = slaves }
	}

	pop_support_factors = {
		movement_support_bureaucrats
		movement_support_officers
		movement_support_clerks
		movement_support_soldiers
		movement_support_ruler_popularity
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_suppressed = yes }
			}
			add = {
				desc = "concept_suppressed"
				value = suppression_base_value
				multiply = {
					value = -0.5
					add = owner.modifier:country_suppression_attraction_factor
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_bolstered = yes }
			}
			add = {
				desc = "concept_bolstered"
				value = bolstering_base_value
				multiply = {
					value = 0.5
					add = owner.modifier:country_bolster_attraction_factor
				}
			}
		}

		add = {
			value = 3
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = { is_pop_type = bureaucrats }
			add = {
				value = 35
				desc = "POP_BUREAUCRATS"
			}
		}
		else_if = {
			limit = { is_pop_type = officers }
			add = {
				value = 20
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = { is_pop_type = clerks }
			add = {
				value = 10
				desc = "POP_CLERKS"
			}
		}
		else_if = {
			limit = { is_pop_type = soldiers }
			add = {
				value = 8
				desc = "POP_SOLDIERS"
			}
		}
		else_if = {
			limit = { is_pop_type = aristocrats }
			add = {
				value = -2
				desc = "POP_ARISTOCRATS"
			}
		}

		if = {
			limit = {
				state = { is_capital = yes }
			}
			multiply = {
				value = 1.5
				desc = "AOC_MOVEMENT_ROYAL_CAPITAL"
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
			if = {
				limit = { is_capital = yes }
				multiply = 3
			}
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { has_law_or_variant = law_type:law_strengthen_the_nobility }
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_OVERMIGHTY_SUBJECTS"
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}

		else_if = {
			limit = {
				OR = {
					is_pop_type = peasants
					is_pop_type = farmers
				}
			}
			add = {
				value = 4
				desc = "LOWER_NO_ICON"
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_royalist
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}

# Civic republicanism of the Renaissance.
# The Florentine Republic restored in 1494 and 1527, Genoa under Andrea Doria
# (1528), the Santa Junta of the Comuneros (1520), the States party of the
# Dutch Republic and the English Commonwealth (1649).
movement_republican = {
	category = movement_category_ideological

	ideology = ideology_republican
	character_ideologies = {
		ideology_republican_leader
		ideology_radical
		ideology_reformer
	}

	creation_trigger = {
		aoc_country_is_monarchy = yes
		has_technology_researched = renaissance
	}
	creation_weight = {
		value = 100
		if = {
			limit = { has_technology_researched = representative_assemblies }
			multiply = 2
		}
		if = {
			limit = {
				any_neighbouring_state = {
					owner = { aoc_country_is_republic = yes }
				}
			}
			multiply = 1.5
		}
		if = {
			limit = { has_law_or_variant = law_type:law_steppe_horde }
			multiply = 0.1
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	disband_trigger = {
		aoc_country_is_republic = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_republican_leader
			has_ideology = ideology:ideology_radical
			scope:interest_group ?= {
				OR = {
					is_interest_group_type = ig_intelligentsia
					is_interest_group_type = ig_petty_bourgeoisie
					is_interest_group_type = ig_industrialists
				}
			}
		}
	}
	character_support_weight = {
		value = 50
		if = {
			limit = { has_ideology = ideology:ideology_republican_leader }
			multiply = 10
		}
		else_if = {
			limit = { has_ideology = ideology:ideology_radical }
			multiply = 4
		}
	}

	pop_support_trigger = {
		NOT = { is_pop_type = slaves }
	}

	pop_support_factors = {
		movement_support_high_urbanization
		movement_support_high_literacy
		movement_support_capitalists
		movement_support_shopkeepers
		movement_support_academics
		movement_support_clerks
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_suppressed = yes }
			}
			add = {
				desc = "concept_suppressed"
				value = suppression_base_value
				multiply = {
					value = -0.5
					add = owner.modifier:country_suppression_attraction_factor
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_bolstered = yes }
			}
			add = {
				desc = "concept_bolstered"
				value = bolstering_base_value
				multiply = {
					value = 0.5
					add = owner.modifier:country_bolster_attraction_factor
				}
			}
		}

		add = {
			value = 2
			desc = "POP_BASE_SUPPORT"
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = { is_pop_type = capitalists }
			add = {
				value = 30
				desc = "POP_CAPITALISTS"
			}
		}
		else_if = {
			limit = { is_pop_type = academics }
			add = {
				value = 30
				desc = "POP_ACADEMICS"
			}
		}
		else_if = {
			limit = { is_pop_type = shopkeepers }
			add = {
				value = 25
				desc = "POP_SHOPKEEPERS"
			}
		}
		else_if = {
			limit = { is_pop_type = clerks }
			add = {
				value = 10
				desc = "POP_CLERKS"
			}
		}
		else_if = {
			limit = { is_pop_type = laborers }
			add = {
				value = 4
				desc = "POP_LABORERS"
			}
		}

		multiply = {
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.1
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = state_urbanization_rate
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { has_law_or_variant = law_type:law_autocracy }
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_TYRANNY"
			}
		}
		if = {
			limit = {
				owner ?= {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_republican
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}
