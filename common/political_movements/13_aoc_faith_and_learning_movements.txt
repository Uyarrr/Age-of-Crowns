# Churchmen defending the privileges and doctrine of the established faith.
# The Council of Trent (1545-1563), the Catholic League (1576), the partisans
# of the Inquisition, the Josephite "possessors" of Muscovy and the Ottoman ulema.
movement_clerical = {
	category = movement_category_ideological

	ideology = ideology_spiritualist
	character_ideologies = {
		ideology_theocrat
		ideology_moderate
	}

	creation_trigger = {
		NOT = { has_law_or_variant = law_type:law_state_atheism }
	}
	creation_weight = {
		value = 100
		if = {
			limit = { aoc_country_has_weakened_church = yes }
			multiply = 3
		}
		if = {
			limit = { has_technology_researched = confessionalism }
			multiply = 1.5
		}
		if = {
			limit = { has_law_or_variant = law_type:law_theocracy }
			multiply = 0.5
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_theocrat
			scope:interest_group ?= {
				OR = {
					is_interest_group_type = ig_devout
					has_ideology = ideology:ideology_spiritualist
				}
			}
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = { has_ideology = ideology:ideology_theocrat }
			multiply = 5
		}
		if = {
			limit = {
				scope:interest_group ?= { is_interest_group_type = ig_devout }
			}
			multiply = 3
		}
	}

	pop_support_trigger = {
		NOT = { is_pop_type = slaves }
		religion = { is_state_religion = root.owner }
	}

	pop_support_factors = {
		movement_support_clergymen
		movement_support_peasants
		movement_support_aristocrats
		movement_support_low_literacy
		movement_support_curtailed_clergy
		movement_support_confessionalism
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_suppressed = yes }
			}
			add = {
				desc = "concept_suppressed"
				value = suppression_base_value
				multiply = {
					value = -0.5
					add = owner.modifier:country_suppression_attraction_factor
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= { is_being_bolstered = yes }
			}
			add = {
				desc = "concept_bolstered"
				value = bolstering_base_value
				multiply = {
					value = 0.5
					add = owner.modifier:country_bolster_attraction_factor
				}
			}
		}

		add = {
			value = 3
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = { is_pop_type = clergymen }
			add = {
				value = 50
				desc = "POP_CLERGYMEN"
			}
		}
		else_if = {
			limit = { is_pop_type = peasants }
			add = {
				value = 8
				desc = "POP_PEASANTS"
			}
		}
		else_if = {
			limit = { is_pop_type = aristocrats }
			add = {
				value = 6
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = { is_pop_type = farmers }
			add = {
				value = 5
				desc = "POP_FARMERS"
			}
		}

		if = {
			limit = {
				owner = { aoc_country_has_weakened_church = yes }
			}
			multiply = {
				value = 1.5
				desc = "AOC_MOVEMENT_CHURCH_UNDER_ATTACK"
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		interest_group_can_join = {
			leader = { religion = scope:political_movement.owner.religion }
		}

		state_weight = {
			value = 1
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= { aoc_country_has_weakened_church = yes }
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_CHURCH_UNDER_ATTACK"
			}
		}
		if = {
			limit = {
				owner.ruler ?= {
					NOT = { religion = root.owner.religion }
				}
			}
			add = {
				value = 0.1
				desc = "RULER_WRONG_RELIGION"
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}

			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_outlawed_dissent
						has_law_or_variant = law_type:law_censorship
					}
				}
			}
			add = {
				value = 0.1
				desc = "AOC_MOVEMENT_CENSORSHIP"
			}
		}
		if = {
			limit = {
				owner ?= {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_enlightenment
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = {
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}
